     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import type { Partition, PlannerDoc, PlannerObject, Pt, Underlay, View } from '@/lib/planner/types'
     5→import { DEFAULT_DOC, uid } from '@/lib/planner/types'
     6→import type { Preset } from '@/lib/planner/presets'
     7→import { getPreset } from '@/lib/planner/presets'
     8→import type { Tool } from '@/lib/planner/tools'
     9→import {
    10→  closestOnSegment,
    11→  distToSegment,
    12→  pointInObject,
    13→  pointInRect,
    14→  pointsBBox,
    15→  objectsBBox,
    16→  unionBBox,
    17→  rotateHandlePos,
    18→  screenToPlan,
    19→  snapObjectPos,
    20→  snapValue,
    21→} from '@/lib/planner/geometry'
    22→import { computeUnderlayPlacement, fileToUnderlaySource, packDocForHistory, unpackDocFromHistory } from '@/lib/planner/underlay'
    23→import { drawScene } from '@/lib/planner/draw'
    24→import { exportJSON, exportPNG, makeSaveFile, validateSaveFile } from '@/lib/planner/export'
    25→import { Catalog } from './Catalog'
    26→import { PropertiesPanel } from './PropertiesPanel'
    27→import { TopBar } from './TopBar'
    28→import { Button } from '@/components/ui/button'
    29→import { toast } from 'sonner'
    30→import {
    31→  PencilLine,
    32→  Ruler,
    33→} from 'lucide-react'
    34→
    35→const STORAGE_KEY = 'room-planner-v1'
    36→const MIN_SCALE = 0.02
    37→const MAX_SCALE = 2
    38→
    39→interface Interaction {
    40→  type: 'none' | 'pan' | 'drag' | 'rotate' | 'vertex' | 'underlayDrag' | 'partitionVertex'
    41→  panStart?: { ox: number; oy: number; px: number; py: number }
    42→  grabDX?: number
    43→  grabDY?: number
    44→  moved?: boolean
    45→  vertexIdx?: number
    46→  partitionId?: string
    47→}
    48→
    49→export function Planner() {
    50→  // ---------- состояние ----------
    51→  const [doc, setDoc] = useState<PlannerDoc>(DEFAULT_DOC)
    52→  const [showGrid, setShowGrid] = useState(true)
    53→  const [tool, setTool] = useState<Tool>('select')
    54→  const [selectedId, setSelectedId] = useState<string | null>(null)
    55→  const [placePreset, setPlacePreset] = useState<Preset | null>(null)
    56→  const [savedAt, setSavedAt] = useState<string | null>(null)
    57→  const [hist, setHist] = useState({ p: 0, f: 0 })
    58→  const [underlaySelected, setUnderlaySelected] = useState(false)
    59→  const [selectedPartitionId, setSelectedPartitionId] = useState<string | null>(null)
    60→
    61→  // ---------- refs ----------
    62→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
    63→  const wrapRef = useRef<HTMLDivElement | null>(null)
    64→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    65→  const underlayInputRef = useRef<HTMLInputElement | null>(null)
    66→  const coordsRef = useRef<HTMLSpanElement | null>(null)
    67→  const zoomRef = useRef<HTMLSpanElement | null>(null)
    68→  const drawRef = useRef<() => void>(() => {})
    69→
    70→  const viewRef = useRef<View>({ scale: 0.3, ox: 250, oy: 180 })
    71→  const sizeRef = useRef({ w: 800, h: 600 })
    72→  const docRef = useRef(doc)
    73→  const toolRef = useRef(tool)
    74→  const selectedIdRef = useRef(selectedId)
    75→  const placePresetRef = useRef(placePreset)
    76→  const showGridRef = useRef(showGrid)
    77→  const drawingPtsRef = useRef<Pt[] | null>(null)
    78→  const cursorPlanRef = useRef<Pt | null>(null)
    79→  const ghostRef = useRef<PlannerObject | null>(null)
    80→  const dragVertexRef = useRef<number | null>(null)
    81→  const spaceRef = useRef(false)
    82→  const userViewRef = useRef(false)
    83→  const loadedRef = useRef(false)
    84→  const lastNudgeRef = useRef(0)
    85→  const underlaySelectedRef = useRef(false)
    86→  const selectedPartitionIdRef = useRef<string | null>(null)
    87→  const partitionVertexRef = useRef<number | null>(null)
    88→  const interRef = useRef<Interaction>({ type: 'none' })
    89→  const historyRef = useRef<{ past: string[]; future: string[] }>({ past: [], future: [] })
    90→
    91→  // ---------- зеркала состояния для слушателей ----------
    92→  useEffect(() => void (docRef.current = doc), [doc])
    93→  useEffect(() => void (toolRef.current = tool), [tool])
    94→  useEffect(() => void (selectedIdRef.current = selectedId), [selectedId])
    95→  useEffect(() => void (placePresetRef.current = placePreset), [placePreset])
    96→  useEffect(() => void (showGridRef.current = showGrid), [showGrid])
    97→  useEffect(() => void (underlaySelectedRef.current = underlaySelected), [underlaySelected])
    98→  useEffect(() => void (selectedPartitionIdRef.current = selectedPartitionId), [selectedPartitionId])
    99→
   100→  // ---------- базовые операции ----------
   101→  const applyDoc = useCallback((next: PlannerDoc) => {
   102→    docRef.current = next
   103→    setDoc(next)
   104→  }, [])
   105→
   106→  const pushHistory = useCallback((snapshot?: PlannerDoc) => {
   107→    const h = historyRef.current
   108→    h.past.push(packDocForHistory(snapshot ?? docRef.current))
   109→    if (h.past.length > 60) h.past.shift()
   110→    h.future = []
   111→    setHist({ p: h.past.length, f: 0 })
   112→  }, [])
   113→
   114→  const undo = useCallback(() => {
   115→    const h = historyRef.current
   116→    if (!h.past.length) return
   117→    h.future.push(packDocForHistory(docRef.current))
   118→    const prev = h.past.pop() as string
   119→    applyDoc(unpackDocFromHistory(prev))
   120→    setHist({ p: h.past.length, f: h.future.length })
   121→  }, [applyDoc])
   122→
   123→  const redo = useCallback(() => {
   124→    const h = historyRef.current
   125→    if (!h.future.length) return
   126→    h.past.push(packDocForHistory(docRef.current))
   127→    const next = h.future.pop() as string
   128→    applyDoc(unpackDocFromHistory(next))
   129→    setHist({ p: h.past.length, f: h.future.length })
   130→  }, [applyDoc])
   131→
   132→  const fitView = useCallback(() => {
   133→    const d = docRef.current
   134→    let bbox = unionBBox(d.room && d.room.length >= 3 ? pointsBBox(d.room) : null, objectsBBox(d.objects))
   135→    bbox = unionBBox(bbox, d.partitions.length > 0 ? pointsBBox(d.partitions.flatMap((p) => p.pts)) : null)
   136→    if (!bbox) bbox = { minX: 0, minY: 0, maxX: 600, maxY: 500 }
   137→    const pad = 60
   138→    const wCm = Math.max(100, bbox.maxX - bbox.minX + pad * 2)
   139→    const hCm = Math.max(100, bbox.maxY - bbox.minY + pad * 2)
   140→    const { w, h } = sizeRef.current
   141→    const scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, Math.min((w - 24) / wCm, (h - 24) / hCm)))
   142→    viewRef.current = {
   143→      scale,
   144→      ox: (w - (bbox.maxX - bbox.minX) * scale) / 2 - bbox.minX * scale,
   145→      oy: (h - (bbox.maxY - bbox.minY) * scale) / 2 - bbox.minY * scale,
   146→    }
   147→  }, [])
   148→
   149→  const draw = useCallback(() => {
   150→    const canvas = canvasRef.current
   151→    if (!canvas) return
   152→    const ctx = canvas.getContext('2d')
   153→    if (!ctx) return
   154→    const dpr = Math.min(window.devicePixelRatio || 1, 2)
   155→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   156→    drawScene(ctx, sizeRef.current.w, sizeRef.current.h, docRef.current, viewRef.current, {
   157→      showGrid: showGridRef.current,
   158→      selectedId: selectedIdRef.current,
   159→      drawingPts: drawingPtsRef.current,
   160→      cursor: cursorPlanRef.current,
   161→      ghost: ghostRef.current,
   162→      draggingVertex: dragVertexRef.current,
   163→      showVertexHandles: toolRef.current === 'select' && !placePresetRef.current,
   164→      underlaySelected: underlaySelectedRef.current,
   165→      selectedPartitionId: selectedPartitionIdRef.current,
   166→      draggingPartitionVertex: partitionVertexRef.current,
   167→      drawingMode: toolRef.current === 'partition' ? 'partition' : 'room',
   168→      onImageLoad: () => drawRef.current(),
   169→    })
   170→    if (zoomRef.current) zoomRef.current.textContent = `${Math.round(viewRef.current.scale * 100)}%`
   171→  }, [])
   172→
   173→  useEffect(() => void (drawRef.current = draw), [draw])
   174→
   175→  const zoomAt = useCallback(
   176→    (px: number, py: number, factor: number) => {
   177→      const v = viewRef.current
   178→      const ns = Math.min(MAX_SCALE, Math.max(MIN_SCALE, v.scale * factor))
   179→      const plan = screenToPlan(px, py, v)
   180→      viewRef.current = { scale: ns, ox: px - plan.x * ns, oy: py - plan.y * ns }
   181→      userViewRef.current = true
   182→      draw()
   183→    },
   184→    [draw],
   185→  )
   186→
   187→  const zoomCenter = useCallback(
   188→    (factor: number) => {
   189→      const { w, h } = sizeRef.current
   190→      zoomAt(w / 2, h / 2, factor)
   191→    },
   192→    [zoomAt],
   193→  )
   194→
   195→  // ---------- размеры канваса ----------
   196→  useEffect(() => {
   197→    const wrap = wrapRef.current
   198→    const canvas = canvasRef.current
   199→    if (!wrap || !canvas) return
   200→    const ro = new ResizeObserver(() => {
   201→      const r = wrap.getBoundingClientRect()
   202→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   203→      sizeRef.current = { w: r.width, h: r.height }
   204→      canvas.width = Math.max(1, Math.round(r.width * dpr))
   205→      canvas.height = Math.max(1, Math.round(r.height * dpr))
   206→      if (!userViewRef.current) fitView()
   207→      draw()
   208→    })
   209→    ro.observe(wrap)
   210→    return () => ro.disconnect()
   211→  }, [draw, fitView])
   212→
   213→  // ---------- загрузка из localStorage ----------
   214→  useEffect(() => {
   215→    try {
   216→      const raw = localStorage.getItem(STORAGE_KEY)
   217→      if (raw) {
   218→        const parsedJson = JSON.parse(raw)
   219→        const parsed = validateSaveFile(parsedJson)
   220→        if (parsed) {
   221→          docRef.current = parsed
   222→          setDoc(parsed)
   223→          if (typeof parsedJson.showGrid === 'boolean') {
   224→            showGridRef.current = parsedJson.showGrid
   225→            setShowGrid(parsedJson.showGrid)
   226→          }
   227→        }
   228→      }
   229→    } catch {
   230→      // повреждённое сохранение — игнорируем
   231→    }
   232→    loadedRef.current = true
   233→  }, [])
   234→
   235→  // ---------- автосохранение ----------
   236→  useEffect(() => {
   237→    if (!loadedRef.current) return
   238→    const t = setTimeout(() => {
   239→      try {
   240→        localStorage.setItem(STORAGE_KEY, JSON.stringify(makeSaveFile(doc, showGrid)))
   241→        setSavedAt(new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' }))
   242→      } catch {
   243→        // нет доступа к localStorage
   244→      }
   245→    }, 500)
   246→    return () => clearTimeout(t)
   247→  }, [doc, showGrid])
   248→
   249→  // ---------- перерисовка при изменении состояния ----------
   250→  useEffect(() => {
   251→    draw()
   252→  }, [doc, showGrid, tool, selectedId, placePreset, underlaySelected, selectedPartitionId, draw])
   253→
   254→  // ---------- очистка выделения ----------
   255→  useEffect(() => {
   256→    if (selectedId && !doc.objects.some((o) => o.id === selectedId)) {
   257→      selectedIdRef.current = null
   258→      setSelectedId(null)
   259→    }
   260→  }, [doc, selectedId])
   261→
   262→  useEffect(() => {
   263→    if (selectedPartitionId && !doc.partitions.some((p) => p.id === selectedPartitionId)) {
   264→      selectedPartitionIdRef.current = null
   265→      setSelectedPartitionId(null)
   266→    }
   267→  }, [doc, selectedPartitionId])
   268→
   269→  // ---------- действия ----------
   270→  const startWallDrawing = useCallback(() => {
   271→    drawingPtsRef.current = []
   272→    cursorPlanRef.current = null
   273→    placePresetRef.current = null
   274→    setPlacePreset(null)
   275→    toolRef.current = 'wall'
   276→    setTool('wall')
   277→    draw()
   278→  }, [draw])
   279→
   280→  const handleToolChange = useCallback(
   281→    (t: Tool) => {
   282→      if (toolRef.current !== t) drawingPtsRef.current = null
   283→      if (t !== 'select') {
   284→        placePresetRef.current = null
   285→        setPlacePreset(null)
   286→        ghostRef.current = null
   287→      }
   288→      toolRef.current = t
   289→      setTool(t)
   290→      draw()
   291→    },
   292→    [draw],
   293→  )
   294→
   295→  const closeRoom = useCallback(() => {
   296→    const pts = drawingPtsRef.current
   297→    if (!pts) return
   298→    if (pts.length < 3) {
   299→      drawingPtsRef.current = null
   300→      draw()
   301→      return
   302→    }
   303→    pushHistory()
   304→    applyDoc({ ...docRef.current, room: pts })
   305→    drawingPtsRef.current = null
   306→    cursorPlanRef.current = null
   307→    toolRef.current = 'select'
   308→    setTool('select')
   309→    userViewRef.current = false
   310→    fitView()
   311→    draw()
   312→    toast.success('Комната создана — добавьте объекты из каталога')
   313→  }, [applyDoc, pushHistory, fitView, draw])
   314→
   315→  /** Завершить рисование перегородки: Enter, клик по последней точке или двойной клик */
   316→  const finishPartition = useCallback(() => {
   317→    const pts = drawingPtsRef.current
   318→    if (!pts) return
   319→    drawingPtsRef.current = null
   320→    cursorPlanRef.current = null
   321→    // убираем дубли точек от прилипания
   322→    const cleaned: Pt[] = []
   323→    for (const p of pts) {
   324→      const q = cleaned[cleaned.length - 1]
   325→      if (!q || Math.hypot(q.x - p.x, q.y - p.y) > 1) cleaned.push(p)
   326→    }
   327→    if (cleaned.length >= 2) {
   328→      pushHistory()
   329→      const part: Partition = { id: uid(), pts: cleaned }
   330→      applyDoc({ ...docRef.current, partitions: [...docRef.current.partitions, part] })
   331→      toast.success('Перегородка добавлена — можно рисовать следующую, Esc — выйти из режима')
   332→    }
   333→    draw()
   334→  }, [applyDoc, pushHistory, draw])
   335→
   336→  const deletePartition = useCallback(
   337→    (id: string) => {
   338→      pushHistory()
   339→      applyDoc({ ...docRef.current, partitions: docRef.current.partitions.filter((p) => p.id !== id) })
   340→      toast('Перегородка удалена', { icon: '🗑️' })
   341→    },
   342→    [applyDoc, pushHistory],
   343→  )
   344→
   345→  const placeObject = useCallback(
   346→    (plan: Pt, keep: boolean) => {
   347→      const preset = placePresetRef.current
   348→      if (!preset) return
   349→      const step = docRef.current.gridStep
   350→      let x = plan.x
   351→      let y = plan.y
   352→      if (showGridRef.current) {
   353→        const s = snapObjectPos({ x, y, w: preset.w, h: preset.h }, step)
   354→        x = s.x
   355→        y = s.y
   356→      }
   357→      const obj: PlannerObject = {
   358→        id: uid(),
   359→        presetId: preset.id,
   360→        name: preset.name,
   361→        x,
   362→        y,
   363→        w: preset.w,
   364→        h: preset.h,
   365→        angle: 0,
   366→        color: preset.color,
   367→      }
   368→      pushHistory()
   369→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, obj] })
   370→      selectedIdRef.current = obj.id
   371→      setSelectedId(obj.id)
   372→      if (!keep) {
   373→        placePresetRef.current = null
   374→        setPlacePreset(null)
   375→        ghostRef.current = null
   376→      }
   377→    },
   378→    [applyDoc, pushHistory],
   379→  )
   380→
   381→  const deleteObject = useCallback(
   382→    (id: string) => {
   383→      pushHistory()
   384→      applyDoc({ ...docRef.current, objects: docRef.current.objects.filter((o) => o.id !== id) })
   385→      toast('Объект удалён', { icon: '🗑️' })
   386→    },
   387→    [applyDoc, pushHistory],
   388→  )
   389→
   390→  const duplicateObject = useCallback(
   391→    (id: string) => {
   392→      const o = docRef.current.objects.find((x) => x.id === id)
   393→      if (!o) return
   394→      pushHistory()
   395→      const copy: PlannerObject = { ...o, id: uid(), x: o.x + 20, y: o.y + 20 }
   396→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, copy] })
   397→      selectedIdRef.current = copy.id
   398→      setSelectedId(copy.id)
   399→    },
   400→    [applyDoc, pushHistory],
   401→  )
   402→
   403→  const updateObject = useCallback(
   404→    (id: string, patch: Partial<PlannerObject>) => {
   405→      applyDoc({ ...docRef.current, objects: docRef.current.objects.map((o) => (o.id === id ? { ...o, ...patch } : o)) })
   406→    },
   407→    [applyDoc],
   408→  )
   409→
   410→  const clearRoom = useCallback(() => {
   411→    if (!docRef.current.room) return
   412→    pushHistory()
   413→    applyDoc({ ...docRef.current, room: null })
   414→    toast('Стены удалены — нарисуйте новый контур')
   415→  }, [applyDoc, pushHistory])
   416→
   417→  // ---------- подложка ----------
   418→  const handleUnderlayFile = useCallback(
   419→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   420→      const file = e.target.files?.[0]
   421→      e.target.value = ''
   422→      if (!file) return
   423→      if (!file.type.startsWith('image/')) {
   424→        toast.error('Выберите файл изображения (PNG, JPG, WebP)')
   425→        return
   426→      }
   427→      try {
   428→        const { src, imgW, imgH, bytes } = await fileToUnderlaySource(file)
   429→        const place = computeUnderlayPlacement(imgW, imgH, docRef.current, sizeRef.current, viewRef.current)
   430→        pushHistory()
   431→        applyDoc({
   432→          ...docRef.current,
   433→          underlay: { src, imgW, imgH, x: place.x, y: place.y, w: place.w, h: place.h, angle: 0, opacity: 0.55, visible: true },
   434→        })
   435→        selectedIdRef.current = null
   436→        setSelectedId(null)
   437→        underlaySelectedRef.current = true
   438→        setUnderlaySelected(true)
   439→        if (bytes > 3 * 1024 * 1024) {
   440→          toast.warning('Подложка добавлена, но файл большой — автосохранение может замедлиться')
   441→        } else {
   442→          toast.success('Подложка добавлена — задайте ей точную ширину в панели справа')
   443→        }
   444→      } catch {
   445→        toast.error('Не удалось прочитать изображение')
   446→      }
   447→    },
   448→    [applyDoc, pushHistory],
   449→  )
   450→
   451→  const updateUnderlay = useCallback(
   452→    (patch: Partial<Underlay>) => {
   453→      const u = docRef.current.underlay
   454→      if (!u) return
   455→      applyDoc({ ...docRef.current, underlay: { ...u, ...patch } })
   456→    },
   457→    [applyDoc],
   458→  )
   459→
   460→  const removeUnderlay = useCallback(() => {
   461→    if (!docRef.current.underlay) return
   462→    pushHistory()
   463→    applyDoc({ ...docRef.current, underlay: null })
   464→    underlaySelectedRef.current = false
   465→    setUnderlaySelected(false)
   466→    toast('Подложка удалена', { icon: '🗑️' })
   467→  }, [applyDoc, pushHistory])
   468→
   469→  const fitUnderlay = useCallback(() => {
   470→    const u = docRef.current.underlay
   471→    if (!u) return
   472→    const place = computeUnderlayPlacement(u.imgW, u.imgH, docRef.current, sizeRef.current, viewRef.current)
   473→    pushHistory()
   474→    applyDoc({ ...docRef.current, underlay: { ...u, ...place } })
   475→  }, [applyDoc, pushHistory])
   476→
   477→  const selectUnderlay = useCallback(() => {
   478→    selectedIdRef.current = null
   479→    setSelectedId(null)
   480→    underlaySelectedRef.current = true
   481→    setUnderlaySelected(true)
   482→    draw()
   483→  }, [draw])
   484→
   485→  const handleNew = useCallback(() => {
   486→    pushHistory()
   487→    applyDoc({ ...DEFAULT_DOC, gridStep: docRef.current.gridStep })
   488→    selectedIdRef.current = null
   489→    setSelectedId(null)
   490→    selectedPartitionIdRef.current = null
   491→    setSelectedPartitionId(null)
   492→    underlaySelectedRef.current = false
   493→    setUnderlaySelected(false)
   494→    drawingPtsRef.current = null
   495→    ghostRef.current = null
   496→    userViewRef.current = false
   497→    fitView()
   498→    draw()
   499→    toast('Создан новый проект')
   500→  }, [applyDoc, pushHistory, fitView, draw])
   501→
   502→  const handleImportFile = useCallback(
   503→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   504→      const file = e.target.files?.[0]
   505→      e.target.value = ''
   506→      if (!file) return
   507→      try {
   508→        const text = await file.text()
   509→        const parsed = validateSaveFile(JSON.parse(text))
   510→        if (!parsed) throw new Error('bad file')
   511→        pushHistory()
   512→        applyDoc(parsed)
   513→        selectedIdRef.current = null
   514→        setSelectedId(null)
   515→        selectedPartitionIdRef.current = null
   516→        setSelectedPartitionId(null)
   517→        underlaySelectedRef.current = false
   518→        setUnderlaySelected(false)
   519→        userViewRef.current = false
   520→        fitView()
   521→        draw()
   522→        toast.success('Проект загружен из файла')
   523→      } catch {
   524→        toast.error('Не удалось прочитать файл проекта')
   525→      }
   526→    },
   527→    [applyDoc, pushHistory, fitView, draw],
   528→  )
   529→
   530→  const handlePickPreset = useCallback(

   562→    const snapPartitionPoint = (plan: Pt): Pt => {
   563→      const s = viewRef.current.scale
   564→      const radV = 10 / s
   565→      let bestVert: Pt | null = null
   566→      let bestVertD = radV
   567→      const verts = [...(docRef.current.room ?? []), ...docRef.current.partitions.flatMap((p) => p.pts)]

   569→        const d = Math.hypot(v.x - plan.x, v.y - plan.y)
   570→        if (d < bestVertD) {
   571→          bestVertD = d
   572→          bestVert = v
   573→        }
   574→      }
   575→      if (bestVert) return bestVert
   576→      const radS = 12 / s
   577→      const segs: [Pt, Pt][] = []
   578→      const room = docRef.current.room
   579→      if (room && room.length >= 3) {
   580→        for (let i = 0; i < room.length; i++) segs.push([room[i], room[(i + 1) % room.length]])
   581→      }
   582→      for (const part of docRef.current.partitions) {
   583→        for (let i = 0; i < part.pts.length - 1; i++) segs.push([part.pts[i], part.pts[i + 1]])
   584→      }
   585→      let bestSeg: Pt | null = null
   586→      let bestSegD = radS
   587→      for (const [a, b] of segs) {
   588→        const c = closestOnSegment(plan, a, b)
   589→        const d = Math.hypot(c.x - plan.x, c.y - plan.y)
   590→        if (d < bestSegD) {
   591→          bestSegD = d
   592→          bestSeg = c
   593→        }
   594→      }
   595→      if (bestSeg) return bestSeg
   596→      const step = docRef.current.gridStep
