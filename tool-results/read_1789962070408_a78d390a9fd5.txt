    32→    32→
    33→    33→const STORAGE_KEY = 'room-planner-v1'
    34→    34→const MIN_SCALE = 0.02
    35→    35→const MAX_SCALE = 2
    36→    36→
    37→    37→interface Interaction {
    38→    38→  type: 'none' | 'pan' | 'drag' | 'rotate' | 'vertex' | 'underlayDrag'
    39→    39→  panStart?: { ox: number; oy: number; px: number; py: number }
    40→    40→  grabDX?: number
    41→    41→  grabDY?: number
    42→    42→  moved?: boolean
    43→    43→  vertexIdx?: number
    44→    44→}
    45→    45→
    46→    46→export function Planner() {
    47→    47→  // ---------- состояние ----------
    48→    48→  const [doc, setDoc] = useState<PlannerDoc>(DEFAULT_DOC)
    49→    49→  const [showGrid, setShowGrid] = useState(true)
    50→    50→  const [tool, setTool] = useState<Tool>('select')
    51→    51→  const [selectedId, setSelectedId] = useState<string | null>(null)
    52→    52→  const [placePreset, setPlacePreset] = useState<Preset | null>(null)
    53→    53→  const [savedAt, setSavedAt] = useState<string | null>(null)
    54→    54→  const [hist, setHist] = useState({ p: 0, f: 0 })
    55→    55→  const [underlaySelected, setUnderlaySelected] = useState(false)
    56→    56→
    57→    57→  // ---------- refs ----------
    58→    58→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
    59→    59→  const wrapRef = useRef<HTMLDivElement | null>(null)
    60→    60→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    61→    61→  const underlayInputRef = useRef<HTMLInputElement | null>(null)
    62→    62→  const coordsRef = useRef<HTMLSpanElement | null>(null)
    63→    63→  const zoomRef = useRef<HTMLSpanElement | null>(null)
    64→    64→  const drawRef = useRef<() => void>(() => {})
    65→    65→
    66→    66→  const viewRef = useRef<View>({ scale: 0.3, ox: 250, oy: 180 })
    67→    67→  const sizeRef = useRef({ w: 800, h: 600 })
    68→    68→  const docRef = useRef(doc)
    69→    69→  const toolRef = useRef(tool)
    70→    70→  const selectedIdRef = useRef(selectedId)
    71→    71→  const placePresetRef = useRef(placePreset)
    72→    72→  const showGridRef = useRef(showGrid)
    73→    73→  const drawingPtsRef = useRef<Pt[] | null>(null)
    74→    74→  const cursorPlanRef = useRef<Pt | null>(null)
    75→    75→  const ghostRef = useRef<PlannerObject | null>(null)
    76→    76→  const dragVertexRef = useRef<number | null>(null)
    77→    77→  const spaceRef = useRef(false)
    78→    78→  const userViewRef = useRef(false)
    79→    79→  const loadedRef = useRef(false)
    80→    80→  const lastNudgeRef = useRef(0)
    81→    81→  const underlaySelectedRef = useRef(false)
    82→    82→  const interRef = useRef<Interaction>({ type: 'none' })
    83→    83→  const historyRef = useRef<{ past: string[]; future: string[] }>({ past: [], future: [] })
    84→    84→
    85→    85→  // ---------- зеркала состояния для слушателей ----------
    86→    86→  useEffect(() => void (docRef.current = doc), [doc])
    87→    87→  useEffect(() => void (toolRef.current = tool), [tool])
    88→    88→  useEffect(() => void (selectedIdRef.current = selectedId), [selectedId])
    89→    89→  useEffect(() => void (placePresetRef.current = placePreset), [placePreset])
    90→    90→  useEffect(() => void (showGridRef.current = showGrid), [showGrid])
    91→    91→  useEffect(() => void (underlaySelectedRef.current = underlaySelected), [underlaySelected])
    92→    92→
    93→    93→  // ---------- базовые операции ----------
    94→    94→  const applyDoc = useCallback((next: PlannerDoc) => {
    95→    95→    docRef.current = next
    96→    96→    setDoc(next)
    97→    97→  }, [])
    98→    98→
    99→    99→  const pushHistory = useCallback((snapshot?: PlannerDoc) => {
   100→   100→    const h = historyRef.current
   101→   101→    h.past.push(packDocForHistory(snapshot ?? docRef.current))
   102→   102→    if (h.past.length > 60) h.past.shift()
   103→   103→    h.future = []
   104→   104→    setHist({ p: h.past.length, f: 0 })
   105→   105→  }, [])
   106→   106→
   107→   107→  const undo = useCallback(() => {
   108→   108→    const h = historyRef.current
   109→   109→    if (!h.past.length) return
   110→   110→    h.future.push(packDocForHistory(docRef.current))
   111→   111→    const prev = h.past.pop() as string
   112→   112→    applyDoc(unpackDocFromHistory(prev))
   113→   113→    setHist({ p: h.past.length, f: h.future.length })
   114→   114→  }, [applyDoc])
   115→   115→
   116→   116→  const redo = useCallback(() => {
   117→   117→    const h = historyRef.current
   118→   118→    if (!h.future.length) return
   119→   119→    h.past.push(packDocForHistory(docRef.current))
   120→   120→    const next = h.future.pop() as string
   121→   121→    applyDoc(unpackDocFromHistory(next))
   122→   122→    setHist({ p: h.past.length, f: h.future.length })
   123→   123→  }, [applyDoc])
   124→   124→
   125→   125→  const fitView = useCallback(() => {
   126→   126→    const d = docRef.current
   127→   127→    let bbox = unionBBox(d.room && d.room.length >= 3 ? pointsBBox(d.room) : null, objectsBBox(d.objects))
   128→   128→    if (!bbox) bbox = { minX: 0, minY: 0, maxX: 600, maxY: 500 }
   129→   129→    const pad = 60
   130→   130→    const wCm = Math.max(100, bbox.maxX - bbox.minX + pad * 2)
   131→   131→    const hCm = Math.max(100, bbox.maxY - bbox.minY + pad * 2)
   132→   132→    const { w, h } = sizeRef.current
   133→   133→    const scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, Math.min((w - 24) / wCm, (h - 24) / hCm)))
   134→   134→    viewRef.current = {
   135→   135→      scale,
   136→   136→      ox: (w - (bbox.maxX - bbox.minX) * scale) / 2 - bbox.minX * scale,
   137→   137→      oy: (h - (bbox.maxY - bbox.minY) * scale) / 2 - bbox.minY * scale,
   138→   138→    }
   139→   139→  }, [])
   140→   140→
   141→   141→  const draw = useCallback(() => {
   142→   142→    const canvas = canvasRef.current
   143→   143→    if (!canvas) return
   144→   144→    const ctx = canvas.getContext('2d')
   145→   145→    if (!ctx) return
   146→   146→    const dpr = Math.min(window.devicePixelRatio || 1, 2)
   147→   147→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   148→   148→    drawScene(ctx, sizeRef.current.w, sizeRef.current.h, docRef.current, viewRef.current, {
   149→   149→      showGrid: showGridRef.current,
   150→   150→      selectedId: selectedIdRef.current,
   151→   151→      drawingPts: drawingPtsRef.current,
   152→   152→      cursor: cursorPlanRef.current,
   153→   153→      ghost: ghostRef.current,
   154→   154→      draggingVertex: dragVertexRef.current,
   155→   155→      showVertexHandles: toolRef.current === 'select' && !placePresetRef.current,
   156→   156→      underlaySelected: underlaySelectedRef.current,
   157→   157→      onImageLoad: () => drawRef.current(),
   158→   158→    })
   159→   159→    if (zoomRef.current) zoomRef.current.textContent = `${Math.round(viewRef.current.scale * 100)}%`
   160→   160→  }, [])
   161→   161→
   162→   162→  useEffect(() => void (drawRef.current = draw), [draw])
   163→   163→
   164→   164→  const zoomAt = useCallback(
   165→   165→    (px: number, py: number, factor: number) => {
   166→   166→      const v = viewRef.current
   167→   167→      const ns = Math.min(MAX_SCALE, Math.max(MIN_SCALE, v.scale * factor))
   168→   168→      const plan = screenToPlan(px, py, v)
   169→   169→      viewRef.current = { scale: ns, ox: px - plan.x * ns, oy: py - plan.y * ns }
   170→   170→      userViewRef.current = true
   171→   171→      draw()
   172→   172→    },
   173→   173→    [draw],
   174→   174→  )
   175→   175→
   176→   176→  const zoomCenter = useCallback(
   177→   177→    (factor: number) => {
   178→   178→      const { w, h } = sizeRef.current
   179→   179→      zoomAt(w / 2, h / 2, factor)
   180→   180→    },
   181→   181→    [zoomAt],
   182→   182→  )
   183→   183→
   184→   184→  // ---------- размеры канваса ----------
   185→   185→  useEffect(() => {
   186→   186→    const wrap = wrapRef.current
   187→   187→    const canvas = canvasRef.current
   188→   188→    if (!wrap || !canvas) return
   189→   189→    const ro = new ResizeObserver(() => {
   190→   190→      const r = wrap.getBoundingClientRect()
   191→   191→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   192→   192→      sizeRef.current = { w: r.width, h: r.height }
   193→   193→      canvas.width = Math.max(1, Math.round(r.width * dpr))
   194→   194→      canvas.height = Math.max(1, Math.round(r.height * dpr))
   195→   195→      if (!userViewRef.current) fitView()
   196→   196→      draw()
   197→   197→    })
   198→   198→    ro.observe(wrap)
   199→   199→    return () => ro.disconnect()
   200→   200→  }, [draw, fitView])
   201→   201→
   202→   202→  // ---------- загрузка из localStorage ----------
   203→   203→  useEffect(() => {
   204→   204→    try {
   205→   205→      const raw = localStorage.getItem(STORAGE_KEY)
   206→   206→      if (raw) {
   207→   207→        const parsedJson = JSON.parse(raw)
   208→   208→        const parsed = validateSaveFile(parsedJson)
   209→   209→        if (parsed) {
   210→   210→          docRef.current = parsed
   211→   211→          setDoc(parsed)
   212→   212→          if (typeof parsedJson.showGrid === 'boolean') {
   213→   213→            showGridRef.current = parsedJson.showGrid
   214→   214→            setShowGrid(parsedJson.showGrid)
   215→   215→          }
   216→   216→        }
   217→   217→      }
   218→   218→    } catch {
   219→   219→      // повреждённое сохранение — игнорируем
   220→   220→    }
   221→   221→    loadedRef.current = true
   222→   222→  }, [])
   223→   223→
   224→   224→  // ---------- автосохранение ----------
   225→   225→  useEffect(() => {
   226→   226→    if (!loadedRef.current) return
   227→   227→    const t = setTimeout(() => {
   228→   228→      try {
   229→   229→        localStorage.setItem(STORAGE_KEY, JSON.stringify(makeSaveFile(doc, showGrid)))
   230→   230→        setSavedAt(new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' }))
   231→   231→      } catch {
   232→   232→        // нет доступа к localStorage
   233→   233→      }
   234→   234→    }, 500)
   235→   235→    return () => clearTimeout(t)
   236→   236→  }, [doc, showGrid])
   237→   237→
   238→   238→  // ---------- перерисовка при изменении состояния ----------
   239→   239→  useEffect(() => {
   240→   240→    draw()
   241→   241→  }, [doc, showGrid, tool, selectedId, placePreset, underlaySelected, draw])
   242→   242→
   243→   243→  // ---------- очистка выделения ----------
   244→   244→  useEffect(() => {
   245→   245→    if (selectedId && !doc.objects.some((o) => o.id === selectedId)) {
   246→   246→      selectedIdRef.current = null
   247→   247→      setSelectedId(null)
   248→   248→    }
   249→   249→  }, [doc, selectedId])
   250→   250→
   251→   251→  // ---------- действия ----------
   252→   252→  const startWallDrawing = useCallback(() => {
   253→   253→    drawingPtsRef.current = []
   254→   254→    cursorPlanRef.current = null
   255→   255→    placePresetRef.current = null
   256→   256→    setPlacePreset(null)
   257→   257→    toolRef.current = 'wall'
   258→   258→    setTool('wall')
   259→   259→    draw()
   260→   260→  }, [draw])
   261→   261→
   262→   262→  const handleToolChange = useCallback(
   263→   263→    (t: Tool) => {
   264→   264→      if (t !== 'wall') drawingPtsRef.current = null
   265→   265→      if (t !== 'select') {
   266→   266→        placePresetRef.current = null
   267→   267→        setPlacePreset(null)
   268→   268→        ghostRef.current = null
   269→   269→      }
   270→   270→      toolRef.current = t
   271→   271→      setTool(t)
   272→   272→      draw()
   273→   273→    },
   274→   274→    [draw],
   275→   275→  )
   276→   276→
   277→   277→  const closeRoom = useCallback(() => {
   278→   278→    const pts = drawingPtsRef.current
   279→   279→    if (!pts) return
   280→   280→    if (pts.length < 3) {
   281→   281→      drawingPtsRef.current = null
   282→   282→      draw()
   283→   283→      return
   284→   284→    }
   285→   285→    pushHistory()
   286→   286→    applyDoc({ ...docRef.current, room: pts })
   287→   287→    drawingPtsRef.current = null
   288→   288→    cursorPlanRef.current = null
   289→   289→    toolRef.current = 'select'
   290→   290→    setTool('select')
   291→   291→    userViewRef.current = false
   292→   292→    fitView()
   293→   293→    draw()
   294→   294→    toast.success('Комната создана — добавьте объекты из каталога')
   295→   295→  }, [applyDoc, pushHistory, fitView, draw])
   296→   296→
   297→   297→  const placeObject = useCallback(
   298→   298→    (plan: Pt, keep: boolean) => {
   299→   299→      const preset = placePresetRef.current
   300→   300→      if (!preset) return
   301→   301→      const step = docRef.current.gridStep
   302→   302→      let x = plan.x
   303→   303→      let y = plan.y
   304→   304→      if (showGridRef.current) {
   305→   305→        const s = snapObjectPos({ x, y, w: preset.w, h: preset.h }, step)
   306→   306→        x = s.x
   307→   307→        y = s.y
   308→   308→      }
   309→   309→      const obj: PlannerObject = {
   310→   310→        id: uid(),
   311→   311→        presetId: preset.id,
   312→   312→        name: preset.name,
   313→   313→        x,
   314→   314→        y,
   315→   315→        w: preset.w,
   316→   316→        h: preset.h,
   317→   317→        angle: 0,
   318→   318→        color: preset.color,
   319→   319→      }
   320→   320→      pushHistory()
   321→   321→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, obj] })
   322→   322→      selectedIdRef.current = obj.id
   323→   323→      setSelectedId(obj.id)
   324→   324→      if (!keep) {
   325→   325→        placePresetRef.current = null
   326→   326→        setPlacePreset(null)
   327→   327→        ghostRef.current = null
   328→   328→      }
   329→   329→    },
   330→   330→    [applyDoc, pushHistory],
   331→   331→  )
   332→   332→
   333→   333→  const deleteObject = useCallback(
   334→   334→    (id: string) => {
   335→   335→      pushHistory()
   336→   336→      applyDoc({ ...docRef.current, objects: docRef.current.objects.filter((o) => o.id !== id) })
   337→   337→      toast('Объект удалён', { icon: '🗑️' })
   338→   338→    },
   339→   339→    [applyDoc, pushHistory],
   340→   340→  )
   341→   341→
   342→   342→  const duplicateObject = useCallback(
   343→   343→    (id: string) => {
   344→   344→      const o = docRef.current.objects.find((x) => x.id === id)
   345→   345→      if (!o) return
   346→   346→      pushHistory()
   347→   347→      const copy: PlannerObject = { ...o, id: uid(), x: o.x + 20, y: o.y + 20 }
   348→   348→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, copy] })
   349→   349→      selectedIdRef.current = copy.id
   350→   350→      setSelectedId(copy.id)
   351→   351→    },
   352→   352→    [applyDoc, pushHistory],
   353→   353→  )
   354→   354→
   355→   355→  const updateObject = useCallback(
   356→   356→    (id: string, patch: Partial<PlannerObject>) => {
   357→   357→      applyDoc({ ...docRef.current, objects: docRef.current.objects.map((o) => (o.id === id ? { ...o, ...patch } : o)) })
   358→   358→    },
   359→   359→    [applyDoc],
   360→   360→  )
   361→   361→
   362→   362→  const clearRoom = useCallback(() => {
   363→   363→    if (!docRef.current.room) return
   364→   364→    pushHistory()
   365→   365→    applyDoc({ ...docRef.current, room: null })
   366→   366→    toast('Стены удалены — нарисуйте новый контур')
   367→   367→  }, [applyDoc, pushHistory])
   368→   368→
   369→   369→  // ---------- подложка ----------
   370→   370→  const handleUnderlayFile = useCallback(
   371→   371→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   372→   372→      const file = e.target.files?.[0]
   373→   373→      e.target.value = ''
   374→   374→      if (!file) return
   375→   375→      if (!file.type.startsWith('image/')) {
   376→   376→        toast.error('Выберите файл изображения (PNG, JPG, WebP)')
   377→   377→        return
   378→   378→      }
   379→   379→      try {
   380→   380→        const { src, imgW, imgH, bytes } = await fileToUnderlaySource(file)
   381→   381→        const place = computeUnderlayPlacement(imgW, imgH, docRef.current, sizeRef.current, viewRef.current)
   382→   382→        pushHistory()
   383→   383→        applyDoc({
   384→   384→          ...docRef.current,
   385→   385→          underlay: { src, imgW, imgH, x: place.x, y: place.y, w: place.w, h: place.h, angle: 0, opacity: 0.55, visible: true },
   386→   386→        })
   387→   387→        selectedIdRef.current = null
   388→   388→        setSelectedId(null)
   389→   389→        underlaySelectedRef.current = true
   390→   390→        setUnderlaySelected(true)
   391→   391→        if (bytes > 3 * 1024 * 1024) {
   392→   392→          toast.warning('Подложка добавлена, но файл большой — автосохранение может замедлиться')
   393→   393→        } else {
   394→   394→          toast.success('Подложка добавлена — задайте ей точную ширину в панели справа')
   395→   395→        }
   396→   396→      } catch {
   397→   397→        toast.error('Не удалось прочитать изображение')
   398→   398→      }
   399→   399→    },
   400→   400→    [applyDoc, pushHistory],
   401→   401→  )
   402→   402→
   403→   403→  const updateUnderlay = useCallback(
   404→   404→    (patch: Partial<Underlay>) => {
   405→   405→      const u = docRef.current.underlay
   406→   406→      if (!u) return
   407→   407→      applyDoc({ ...docRef.current, underlay: { ...u, ...patch } })
   408→   408→    },
   409→   409→    [applyDoc],
   410→   410→  )
   411→   411→
   412→   412→  const removeUnderlay = useCallback(() => {
   413→   413→    if (!docRef.current.underlay) return
   414→   414→    pushHistory()
   415→   415→    applyDoc({ ...docRef.current, underlay: null })
   416→   416→    underlaySelectedRef.current = false
   417→   417→    setUnderlaySelected(false)
   418→   418→    toast('Подложка удалена', { icon: '🗑️' })
   419→   419→  }, [applyDoc, pushHistory])
   420→   420→
   421→   421→  const fitUnderlay = useCallback(() => {
   422→   422→    const u = docRef.current.underlay
   423→   423→    if (!u) return
   424→   424→    const place = computeUnderlayPlacement(u.imgW, u.imgH, docRef.current, sizeRef.current, viewRef.current)
   425→   425→    pushHistory()
   426→   426→    applyDoc({ ...docRef.current, underlay: { ...u, ...place } })
   427→   427→  }, [applyDoc, pushHistory])
   428→   428→
   429→   429→  const selectUnderlay = useCallback(() => {
   430→   430→    selectedIdRef.current = null
   431→   431→    setSelectedId(null)
   432→   432→    underlaySelectedRef.current = true
   433→   433→    setUnderlaySelected(true)
   434→   434→    draw()
   435→   435→  }, [draw])
   436→   436→
   437→   437→  const handleNew = useCallback(() => {
   438→   438→    pushHistory()
   439→   439→    applyDoc({ ...DEFAULT_DOC, gridStep: docRef.current.gridStep })
   440→   440→    selectedIdRef.current = null
   441→   441→    setSelectedId(null)
   442→   442→    underlaySelectedRef.current = false
   443→   443→    setUnderlaySelected(false)
   444→   444→    drawingPtsRef.current = null
   445→   445→    ghostRef.current = null
   446→   446→    userViewRef.current = false
   447→   447→    fitView()
