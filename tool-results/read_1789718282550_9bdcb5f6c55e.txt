    31→    31→const STORAGE_KEY = 'room-planner-v1'
    32→    32→const MIN_SCALE = 0.02
    33→    33→const MAX_SCALE = 2
    34→    34→
    35→    35→interface Interaction {

    43→    43→
    44→    44→export function Planner() {
    45→    45→  // ---------- состояние ----------
    46→    46→  const [doc, setDoc] = useState<PlannerDoc>(DEFAULT_DOC)
    47→    47→  const [showGrid, setShowGrid] = useState(true)
    48→    48→  const [tool, setTool] = useState<Tool>('select')
    49→    49→  const [selectedId, setSelectedId] = useState<string | null>(null)
    50→    50→  const [placePreset, setPlacePreset] = useState<Preset | null>(null)
    51→    51→  const [savedAt, setSavedAt] = useState<string | null>(null)
    52→    52→  const [hist, setHist] = useState({ p: 0, f: 0 })
    53→    53→
    54→    54→  // ---------- refs ----------
    55→    55→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
    56→    56→  const wrapRef = useRef<HTMLDivElement | null>(null)
    57→    57→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    58→    58→  const coordsRef = useRef<HTMLSpanElement | null>(null)
    59→    59→  const zoomRef = useRef<HTMLSpanElement | null>(null)
    60→    60→
    61→    61→  const viewRef = useRef<View>({ scale: 0.3, ox: 250, oy: 180 })
    62→    62→  const sizeRef = useRef({ w: 800, h: 600 })
    63→    63→  const docRef = useRef(doc)
    64→    64→  const toolRef = useRef(tool)
    65→    65→  const selectedIdRef = useRef(selectedId)
    66→    66→  const placePresetRef = useRef(placePreset)
    67→    67→  const showGridRef = useRef(showGrid)
    68→    68→  const drawingPtsRef = useRef<Pt[] | null>(null)
    69→    69→  const cursorPlanRef = useRef<Pt | null>(null)
    70→    70→  const ghostRef = useRef<PlannerObject | null>(null)
    71→    71→  const dragVertexRef = useRef<number | null>(null)
    72→    72→  const spaceRef = useRef(false)
    73→    73→  const userViewRef = useRef(false)
    74→    74→  const loadedRef = useRef(false)
    75→    75→  const lastNudgeRef = useRef(0)
    76→    76→  const interRef = useRef<Interaction>({ type: 'none' })
    77→    77→  const historyRef = useRef<{ past: string[]; future: string[] }>({ past: [], future: [] })
    78→    78→
    79→    79→  // ---------- зеркала состояния для слушателей ----------
    80→    80→  useEffect(() => void (docRef.current = doc), [doc])
    81→    81→  useEffect(() => void (toolRef.current = tool), [tool])
    82→    82→  useEffect(() => void (selectedIdRef.current = selectedId), [selectedId])
    83→    83→  useEffect(() => void (placePresetRef.current = placePreset), [placePreset])
    84→    84→  useEffect(() => void (showGridRef.current = showGrid), [showGrid])
    85→    85→
    86→    86→  // ---------- базовые операции ----------
    87→    87→  const applyDoc = useCallback((next: PlannerDoc) => {
    88→    88→    docRef.current = next
    89→    89→    setDoc(next)
    90→    90→  }, [])
    91→    91→
    92→    92→  const pushHistory = useCallback((snapshot?: PlannerDoc) => {
    93→    93→    const h = historyRef.current
    94→    94→    h.past.push(JSON.stringify(snapshot ?? docRef.current))
    95→    95→    if (h.past.length > 60) h.past.shift()
    96→    96→    h.future = []
    97→    97→    setHist({ p: h.past.length, f: 0 })
    98→    98→  }, [])
    99→    99→
   100→   100→  const undo = useCallback(() => {
   101→   101→    const h = historyRef.current
   102→   102→    if (!h.past.length) return
   103→   103→    h.future.push(JSON.stringify(docRef.current))
   104→   104→    const prev = h.past.pop() as string
   105→   105→    applyDoc(JSON.parse(prev))
   106→   106→    setHist({ p: h.past.length, f: h.future.length })
   107→   107→  }, [applyDoc])
   108→   108→
   109→   109→  const redo = useCallback(() => {
   110→   110→    const h = historyRef.current
   111→   111→    if (!h.future.length) return
   112→   112→    h.past.push(JSON.stringify(docRef.current))
   113→   113→    const next = h.future.pop() as string
   114→   114→    applyDoc(JSON.parse(next))
   115→   115→    setHist({ p: h.past.length, f: h.future.length })
   116→   116→  }, [applyDoc])
   117→   117→
   118→   118→  const fitView = useCallback(() => {
   119→   119→    const d = docRef.current
   120→   120→    let bbox = unionBBox(d.room && d.room.length >= 3 ? pointsBBox(d.room) : null, objectsBBox(d.objects))
   121→   121→    if (!bbox) bbox = { minX: 0, minY: 0, maxX: 600, maxY: 500 }
   122→   122→    const pad = 60
   123→   123→    const wCm = Math.max(100, bbox.maxX - bbox.minX + pad * 2)
   124→   124→    const hCm = Math.max(100, bbox.maxY - bbox.minY + pad * 2)
   125→   125→    const { w, h } = sizeRef.current
   126→   126→    const scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, Math.min((w - 24) / wCm, (h - 24) / hCm)))
   127→   127→    viewRef.current = {
   128→   128→      scale,
   129→   129→      ox: (w - (bbox.maxX - bbox.minX) * scale) / 2 - bbox.minX * scale,
   130→   130→      oy: (h - (bbox.maxY - bbox.minY) * scale) / 2 - bbox.minY * scale,
   131→   131→    }
   132→   132→  }, [])
   133→   133→
   134→   134→  const draw = useCallback(() => {
   135→   135→    const canvas = canvasRef.current
   136→   136→    if (!canvas) return
   137→   137→    const ctx = canvas.getContext('2d')
   138→   138→    if (!ctx) return
   139→   139→    const dpr = Math.min(window.devicePixelRatio || 1, 2)
   140→   140→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   141→   141→    drawScene(ctx, sizeRef.current.w, sizeRef.current.h, docRef.current, viewRef.current, {
   142→   142→      showGrid: showGridRef.current,
   143→   143→      selectedId: selectedIdRef.current,
   144→   144→      drawingPts: drawingPtsRef.current,
   145→   145→      cursor: cursorPlanRef.current,
   146→   146→      ghost: ghostRef.current,
   147→   147→      draggingVertex: dragVertexRef.current,
   148→   148→      showVertexHandles: toolRef.current === 'select' && !placePresetRef.current,
   149→   149→    })
   150→   150→    if (zoomRef.current) zoomRef.current.textContent = `${Math.round(viewRef.current.scale * 100)}%`
   151→   151→  }, [])
   152→   152→
   153→   153→  const zoomAt = useCallback(
   154→   154→    (px: number, py: number, factor: number) => {
   155→   155→      const v = viewRef.current
   156→   156→      const ns = Math.min(MAX_SCALE, Math.max(MIN_SCALE, v.scale * factor))
   157→   157→      const plan = screenToPlan(px, py, v)
   158→   158→      viewRef.current = { scale: ns, ox: px - plan.x * ns, oy: py - plan.y * ns }
   159→   159→      userViewRef.current = true
   160→   160→      draw()
   161→   161→    },
   162→   162→    [draw],
   163→   163→  )
   164→   164→
   165→   165→  const zoomCenter = useCallback(
   166→   166→    (factor: number) => {
   167→   167→      const { w, h } = sizeRef.current
   168→   168→      zoomAt(w / 2, h / 2, factor)
   169→   169→    },
   170→   170→    [zoomAt],
   171→   171→  )
   172→   172→
   173→   173→  // ---------- размеры канваса ----------
   174→   174→  useEffect(() => {
   175→   175→    const wrap = wrapRef.current
   176→   176→    const canvas = canvasRef.current
   177→   177→    if (!wrap || !canvas) return
   178→   178→    const ro = new ResizeObserver(() => {
   179→   179→      const r = wrap.getBoundingClientRect()
   180→   180→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   181→   181→      sizeRef.current = { w: r.width, h: r.height }
   182→   182→      canvas.width = Math.max(1, Math.round(r.width * dpr))
   183→   183→      canvas.height = Math.max(1, Math.round(r.height * dpr))
   184→   184→      if (!userViewRef.current) fitView()
   185→   185→      draw()
   186→   186→    })
   187→   187→    ro.observe(wrap)
   188→   188→    return () => ro.disconnect()
   189→   189→  }, [draw, fitView])
   190→   190→
   191→   191→  // ---------- загрузка из localStorage ----------
   192→   192→  useEffect(() => {
   193→   193→    try {
   194→   194→      const raw = localStorage.getItem(STORAGE_KEY)
   195→   195→      if (raw) {
   196→   196→        const parsedJson = JSON.parse(raw)
   197→   197→        const parsed = validateSaveFile(parsedJson)
   198→   198→        if (parsed) {
   199→   199→          docRef.current = parsed
   200→   200→          setDoc(parsed)
   201→   201→          if (typeof parsedJson.showGrid === 'boolean') {
   202→   202→            showGridRef.current = parsedJson.showGrid
   203→   203→            setShowGrid(parsedJson.showGrid)
   204→   204→          }
   205→   205→        }
   206→   206→      }
   207→   207→    } catch {
   208→   208→      // повреждённое сохранение — игнорируем
   209→   209→    }
   210→   210→    loadedRef.current = true
   211→   211→  }, [])
   212→   212→
   213→   213→  // ---------- автосохранение ----------
   214→   214→  useEffect(() => {
   215→   215→    if (!loadedRef.current) return
   216→   216→    const t = setTimeout(() => {
   217→   217→      try {
   218→   218→        localStorage.setItem(STORAGE_KEY, JSON.stringify(makeSaveFile(doc, showGrid)))
   219→   219→        setSavedAt(new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' }))
   220→   220→      } catch {
   221→   221→        // нет доступа к localStorage
   222→   222→      }
   223→   223→    }, 500)
   224→   224→    return () => clearTimeout(t)
   225→   225→  }, [doc, showGrid])
   226→   226→
   227→   227→  // ---------- перерисовка при изменении состояния ----------
   228→   228→  useEffect(() => {
   229→   229→    draw()
   230→   230→  }, [doc, showGrid, tool, selectedId, placePreset, draw])
   231→   231→
   232→   232→  // ---------- очистка выделения ----------
   233→   233→  useEffect(() => {
   234→   234→    if (selectedId && !doc.objects.some((o) => o.id === selectedId)) {
   235→   235→      selectedIdRef.current = null
   236→   236→      setSelectedId(null)
   237→   237→    }
   238→   238→  }, [doc, selectedId])
   239→   239→
   240→   240→  // ---------- действия ----------
   241→   241→  const startWallDrawing = useCallback(() => {
   242→   242→    drawingPtsRef.current = []
   243→   243→    cursorPlanRef.current = null
   244→   244→    placePresetRef.current = null
   245→   245→    setPlacePreset(null)
   246→   246→    toolRef.current = 'wall'
   247→   247→    setTool('wall')
   248→   248→    draw()
   249→   249→  }, [draw])
   250→   250→
   251→   251→  const handleToolChange = useCallback(
   252→   252→    (t: Tool) => {
   253→   253→      if (t !== 'wall') drawingPtsRef.current = null
   254→   254→      if (t !== 'select') {
   255→   255→        placePresetRef.current = null
   256→   256→        setPlacePreset(null)
   257→   257→        ghostRef.current = null
   258→   258→      }
   259→   259→      toolRef.current = t
   260→   260→      setTool(t)
   261→   261→      draw()
   262→   262→    },
   263→   263→    [draw],
   264→   264→  )
   265→   265→
   266→   266→  const closeRoom = useCallback(() => {
   267→   267→    const pts = drawingPtsRef.current
   268→   268→    if (!pts) return
   269→   269→    if (pts.length < 3) {
   270→   270→      drawingPtsRef.current = null
   271→   271→      draw()
   272→   272→      return
   273→   273→    }
   274→   274→    pushHistory()
   275→   275→    applyDoc({ ...docRef.current, room: pts })
   276→   276→    drawingPtsRef.current = null
   277→   277→    cursorPlanRef.current = null
   278→   278→    toolRef.current = 'select'
   279→   279→    setTool('select')
   280→   280→    userViewRef.current = false
   281→   281→    fitView()
   282→   282→    draw()
   283→   283→    toast.success('Комната создана — добавьте объекты из каталога')
   284→   284→  }, [applyDoc, pushHistory, fitView, draw])
   285→   285→
   286→   286→  const placeObject = useCallback(
   287→   287→    (plan: Pt, keep: boolean) => {
   288→   288→      const preset = placePresetRef.current
   289→   289→      if (!preset) return
   290→   290→      const step = docRef.current.gridStep
   291→   291→      let x = plan.x
   292→   292→      let y = plan.y
   293→   293→      if (showGridRef.current) {
   294→   294→        const s = snapObjectPos({ x, y, w: preset.w, h: preset.h }, step)
   295→   295→        x = s.x
   296→   296→        y = s.y
   297→   297→      }
   298→   298→      const obj: PlannerObject = {
   299→   299→        id: uid(),
   300→   300→        presetId: preset.id,
   301→   301→        name: preset.name,
   302→   302→        x,
   303→   303→        y,
   304→   304→        w: preset.w,
   305→   305→        h: preset.h,
   306→   306→        angle: 0,
   307→   307→        color: preset.color,
   308→   308→      }
   309→   309→      pushHistory()
   310→   310→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, obj] })
   311→   311→      selectedIdRef.current = obj.id
   312→   312→      setSelectedId(obj.id)
   313→   313→      if (!keep) {
   314→   314→        placePresetRef.current = null
   315→   315→        setPlacePreset(null)
   316→   316→        ghostRef.current = null
   317→   317→      }
   318→   318→    },
   319→   319→    [applyDoc, pushHistory],
   320→   320→  )
   321→   321→
   322→   322→  const deleteObject = useCallback(
   323→   323→    (id: string) => {
   324→   324→      pushHistory()
   325→   325→      applyDoc({ ...docRef.current, objects: docRef.current.objects.filter((o) => o.id !== id) })
   326→   326→      toast('Объект удалён', { icon: '🗑️' })
   327→   327→    },
   328→   328→    [applyDoc, pushHistory],
   329→   329→  )
   330→   330→
   331→   331→  const duplicateObject = useCallback(
   332→   332→    (id: string) => {
   333→   333→      const o = docRef.current.objects.find((x) => x.id === id)
   334→   334→      if (!o) return
   335→   335→      pushHistory()
   336→   336→      const copy: PlannerObject = { ...o, id: uid(), x: o.x + 20, y: o.y + 20 }
   337→   337→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, copy] })
   338→   338→      selectedIdRef.current = copy.id
   339→   339→      setSelectedId(copy.id)
   340→   340→    },
   341→   341→    [applyDoc, pushHistory],
   342→   342→  )
   343→   343→
   344→   344→  const updateObject = useCallback(
   345→   345→    (id: string, patch: Partial<PlannerObject>) => {
   346→   346→      applyDoc({ ...docRef.current, objects: docRef.current.objects.map((o) => (o.id === id ? { ...o, ...patch } : o)) })
   347→   347→    },
   348→   348→    [applyDoc],
   349→   349→  )
   350→   350→
   351→   351→  const clearRoom = useCallback(() => {
   352→   352→    if (!docRef.current.room) return
   353→   353→    pushHistory()
   354→   354→    applyDoc({ ...docRef.current, room: null })
   355→   355→    toast('Стены удалены — нарисуйте новый контур')
   356→   356→  }, [applyDoc, pushHistory])
   357→   357→
   358→   358→  const handleNew = useCallback(() => {
   359→   359→    pushHistory()
   360→   360→    applyDoc({ ...DEFAULT_DOC, gridStep: docRef.current.gridStep })
   361→   361→    selectedIdRef.current = null
   362→   362→    setSelectedId(null)
   363→   363→    drawingPtsRef.current = null
   364→   364→    ghostRef.current = null
   365→   365→    userViewRef.current = false
   366→   366→    fitView()
   367→   367→    draw()
   368→   368→    toast('Создан новый проект')
   369→   369→  }, [applyDoc, pushHistory, fitView, draw])
   370→   370→
   371→   371→  const handleImportFile = useCallback(
   372→   372→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   373→   373→      const file = e.target.files?.[0]
   374→   374→      e.target.value = ''
   375→   375→      if (!file) return
   376→   376→      try {
   377→   377→        const text = await file.text()
   378→   378→        const parsed = validateSaveFile(JSON.parse(text))
   379→   379→        if (!parsed) throw new Error('bad file')
   380→   380→        pushHistory()
   381→   381→        applyDoc(parsed)
   382→   382→        selectedIdRef.current = null
   383→   383→        setSelectedId(null)
   384→   384→        userViewRef.current = false
   385→   385→        fitView()
   386→   386→        draw()
   387→   387→        toast.success('Проект загружен из файла')
   388→   388→      } catch {
   389→   389→        toast.error('Не удалось прочитать файл проекта')
   390→   390→      }
   391→   391→    },
   392→   392→    [applyDoc, pushHistory, fitView, draw],
   393→   393→  )
   394→   394→
   395→   395→  const handlePickPreset = useCallback(
   396→   396→    (p: Preset) => {
   397→   397→      placePresetRef.current = p
   398→   398→      setPlacePreset(p)
   399→   399→      drawingPtsRef.current = null
   400→   400→      toolRef.current = 'select'
   401→   401→      setTool('select')
   402→   402→      draw()
   403→   403→    },
   404→   404→    [draw],
   405→   405→  )
   406→   406→
   407→   407→  // ---------- слушатели мыши и клавиатуры ----------
   408→   408→  useEffect(() => {
   409→   409→    const canvas = canvasRef.current
   410→   410→    if (!canvas) return
   411→   411→
   412→   412→    const getPos = (e: PointerEvent | WheelEvent) => {
   413→   413→      const r = canvas.getBoundingClientRect()
   414→   414→      return { x: e.clientX - r.left, y: e.clientY - r.top }
   415→   415→    }
   416→   416→
   417→   417→    const updateCoords = (plan: Pt | null) => {
   418→   418→      if (!coordsRef.current) return
   419→   419→      coordsRef.current.textContent = plan
   420→   420→        ? `X: ${Math.round(plan.x)} · Y: ${Math.round(plan.y)} см`
   421→   421→        : '—'
   422→   422→    }
   423→   423→
   424→   424→    const updateCursorStyle = (plan: Pt | null) => {
   425→   425→      if (spaceRef.current || toolRef.current === 'pan' || interRef.current.type === 'pan') {
   426→   426→        canvas.style.cursor = interRef.current.type === 'pan' ? 'grabbing' : 'grab'
   427→   427→        return
   428→   428→      }
   429→   429→      if (toolRef.current === 'wall' || placePresetRef.current) {
   430→   430→        canvas.style.cursor = 'crosshair'
   431→   431→        return
   432→   432→      }
   433→   433→      if (plan) {
   434→   434→        const sel = docRef.current.objects.find((o) => o.id === selectedIdRef.current)
   435→   435→        if (sel) {
   436→   436→          const hp = rotateHandlePos(sel, 26 / viewRef.current.scale)
   437→   437→          if (Math.hypot((hp.x - plan.x) * viewRef.current.scale, (hp.y - plan.y) * viewRef.current.scale) < 10) {
   438→   438→            canvas.style.cursor = 'pointer'
   439→   439→            return
   440→   440→          }
   441→   441→        }
   442→   442→        if (docRef.current.room) {
   443→   443→          for (const p of docRef.current.room) {
   444→   444→            if (Math.hypot((p.x - plan.x) * viewRef.current.scale, (p.y - plan.y) * viewRef.current.scale) < 9) {
   445→   445→              canvas.style.cursor = 'pointer'
   446→   446→              return
   447→   447→            }
   448→   448→          }
   449→   449→        }
   450→   450→        for (let i = docRef.current.objects.length - 1; i >= 0; i--) {
   451→   451→          if (pointInObject(docRef.current.objects[i], plan)) {
   452→   452→            canvas.style.cursor = 'move'
   453→   453→            return
   454→   454→          }
   455→   455→        }
   456→   456→      }
