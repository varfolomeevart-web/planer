     1→'use client'
     2→
     3→import { useCallback, useEffect, useRef, useState } from 'react'
     4→import type { PlannerDoc, PlannerObject, Pt, View } from '@/lib/planner/types'
     5→import { DEFAULT_DOC, uid } from '@/lib/planner/types'
     6→import type { Preset } from '@/lib/planner/presets'
     7→import { getPreset } from '@/lib/planner/presets'
     8→import type { Tool } from '@/lib/planner/tools'
     9→import {
    10→  pointInObject,
    11→  pointsBBox,
    12→  objectsBBox,
    13→  unionBBox,
    14→  rotateHandlePos,
    15→  screenToPlan,
    16→  snapObjectPos,
    17→  snapValue,
    18→} from '@/lib/planner/geometry'
    19→import { drawScene } from '@/lib/planner/draw'
    20→import { exportJSON, exportPNG, makeSaveFile, validateSaveFile } from '@/lib/planner/export'
    21→import { Catalog } from './Catalog'
    22→import { PropertiesPanel } from './PropertiesPanel'
    23→import { TopBar } from './TopBar'
    24→import { Button } from '@/components/ui/button'
    25→import { toast } from 'sonner'
    26→import {
    27→  PencilLine,
    28→  Ruler,
    29→} from 'lucide-react'
    30→
    31→const STORAGE_KEY = 'room-planner-v1'
    32→const MIN_SCALE = 0.02
    33→const MAX_SCALE = 2
    34→
    35→interface Interaction {
    36→  type: 'none' | 'pan' | 'drag' | 'rotate' | 'vertex'
    37→  panStart?: { ox: number; oy: number; px: number; py: number }
    38→  grabDX?: number
    39→  grabDY?: number
    40→  moved?: boolean
    41→  vertexIdx?: number
    42→}
    43→
    44→export function Planner() {
    45→  // ---------- состояние ----------
    46→  const [doc, setDoc] = useState<PlannerDoc>(DEFAULT_DOC)
    47→  const [showGrid, setShowGrid] = useState(true)
    48→  const [tool, setTool] = useState<Tool>('select')
    49→  const [selectedId, setSelectedId] = useState<string | null>(null)
    50→  const [placePreset, setPlacePreset] = useState<Preset | null>(null)
    51→  const [savedAt, setSavedAt] = useState<string | null>(null)
    52→  const [hist, setHist] = useState({ p: 0, f: 0 })
    53→
    54→  // ---------- refs ----------
    55→  const canvasRef = useRef<HTMLCanvasElement | null>(null)
    56→  const wrapRef = useRef<HTMLDivElement | null>(null)
    57→  const fileInputRef = useRef<HTMLInputElement | null>(null)
    58→  const coordsRef = useRef<HTMLSpanElement | null>(null)
    59→  const zoomRef = useRef<HTMLSpanElement | null>(null)
    60→
    61→  const viewRef = useRef<View>({ scale: 0.3, ox: 250, oy: 180 })
    62→  const sizeRef = useRef({ w: 800, h: 600 })
    63→  const docRef = useRef(doc)
    64→  const toolRef = useRef(tool)
    65→  const selectedIdRef = useRef(selectedId)
    66→  const placePresetRef = useRef(placePreset)
    67→  const showGridRef = useRef(showGrid)
    68→  const drawingPtsRef = useRef<Pt[] | null>(null)
    69→  const cursorPlanRef = useRef<Pt | null>(null)
    70→  const ghostRef = useRef<PlannerObject | null>(null)
    71→  const dragVertexRef = useRef<number | null>(null)
    72→  const spaceRef = useRef(false)
    73→  const userViewRef = useRef(false)
    74→  const loadedRef = useRef(false)
    75→  const lastNudgeRef = useRef(0)
    76→  const interRef = useRef<Interaction>({ type: 'none' })
    77→  const historyRef = useRef<{ past: string[]; future: string[] }>({ past: [], future: [] })
    78→
    79→  // ---------- зеркала состояния для слушателей ----------
    80→  useEffect(() => void (docRef.current = doc), [doc])
    81→  useEffect(() => void (toolRef.current = tool), [tool])
    82→  useEffect(() => void (selectedIdRef.current = selectedId), [selectedId])
    83→  useEffect(() => void (placePresetRef.current = placePreset), [placePreset])
    84→  useEffect(() => void (showGridRef.current = showGrid), [showGrid])
    85→
    86→  // ---------- базовые операции ----------
    87→  const applyDoc = useCallback((next: PlannerDoc) => {
    88→    docRef.current = next
    89→    setDoc(next)
    90→  }, [])
    91→
    92→  const pushHistory = useCallback((snapshot?: PlannerDoc) => {
    93→    const h = historyRef.current
    94→    h.past.push(JSON.stringify(snapshot ?? docRef.current))
    95→    if (h.past.length > 60) h.past.shift()
    96→    h.future = []
    97→    setHist({ p: h.past.length, f: 0 })
    98→  }, [])
    99→
   100→  const undo = useCallback(() => {
   101→    const h = historyRef.current
   102→    if (!h.past.length) return
   103→    h.future.push(JSON.stringify(docRef.current))
   104→    const prev = h.past.pop() as string
   105→    applyDoc(JSON.parse(prev))
   106→    setHist({ p: h.past.length, f: h.future.length })
   107→  }, [applyDoc])
   108→
   109→  const redo = useCallback(() => {
   110→    const h = historyRef.current
   111→    if (!h.future.length) return
   112→    h.past.push(JSON.stringify(docRef.current))
   113→    const next = h.future.pop() as string
   114→    applyDoc(JSON.parse(next))
   115→    setHist({ p: h.past.length, f: h.future.length })
   116→  }, [applyDoc])
   117→
   118→  const fitView = useCallback(() => {
   119→    const d = docRef.current
   120→    let bbox = unionBBox(d.room && d.room.length >= 3 ? pointsBBox(d.room) : null, objectsBBox(d.objects))
   121→    if (!bbox) bbox = { minX: 0, minY: 0, maxX: 600, maxY: 500 }
   122→    const pad = 60
   123→    const wCm = Math.max(100, bbox.maxX - bbox.minX + pad * 2)
   124→    const hCm = Math.max(100, bbox.maxY - bbox.minY + pad * 2)
   125→    const { w, h } = sizeRef.current
   126→    const scale = Math.min(MAX_SCALE, Math.max(MIN_SCALE, Math.min((w - 24) / wCm, (h - 24) / hCm)))
   127→    viewRef.current = {
   128→      scale,
   129→      ox: (w - (bbox.maxX - bbox.minX) * scale) / 2 - bbox.minX * scale,
   130→      oy: (h - (bbox.maxY - bbox.minY) * scale) / 2 - bbox.minY * scale,
   131→    }
   132→  }, [])
   133→
   134→  const draw = useCallback(() => {
   135→    const canvas = canvasRef.current
   136→    if (!canvas) return
   137→    const ctx = canvas.getContext('2d')
   138→    if (!ctx) return
   139→    const dpr = Math.min(window.devicePixelRatio || 1, 2)
   140→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   141→    drawScene(ctx, sizeRef.current.w, sizeRef.current.h, docRef.current, viewRef.current, {
   142→      showGrid: showGridRef.current,
   143→      selectedId: selectedIdRef.current,
   144→      drawingPts: drawingPtsRef.current,
   145→      cursor: cursorPlanRef.current,
   146→      ghost: ghostRef.current,
   147→      draggingVertex: dragVertexRef.current,
   148→      showVertexHandles: toolRef.current === 'select' && !placePresetRef.current,
   149→    })
   150→    if (zoomRef.current) zoomRef.current.textContent = `${Math.round(viewRef.current.scale * 100)}%`
   151→  }, [])
   152→
   153→  const zoomAt = useCallback(
   154→    (px: number, py: number, factor: number) => {
   155→      const v = viewRef.current
   156→      const ns = Math.min(MAX_SCALE, Math.max(MIN_SCALE, v.scale * factor))
   157→      const plan = screenToPlan(px, py, v)
   158→      viewRef.current = { scale: ns, ox: px - plan.x * ns, oy: py - plan.y * ns }
   159→      userViewRef.current = true
   160→      draw()
   161→    },
   162→    [draw],
   163→  )
   164→
   165→  const zoomCenter = useCallback(
   166→    (factor: number) => {
   167→      const { w, h } = sizeRef.current
   168→      zoomAt(w / 2, h / 2, factor)
   169→    },
   170→    [zoomAt],
   171→  )
   172→
   173→  // ---------- размеры канваса ----------
   174→  useEffect(() => {
   175→    const wrap = wrapRef.current
   176→    const canvas = canvasRef.current
   177→    if (!wrap || !canvas) return
   178→    const ro = new ResizeObserver(() => {
   179→      const r = wrap.getBoundingClientRect()
   180→      const dpr = Math.min(window.devicePixelRatio || 1, 2)
   181→      sizeRef.current = { w: r.width, h: r.height }
   182→      canvas.width = Math.max(1, Math.round(r.width * dpr))
   183→      canvas.height = Math.max(1, Math.round(r.height * dpr))
   184→      if (!userViewRef.current) fitView()
   185→      draw()
   186→    })
   187→    ro.observe(wrap)
   188→    return () => ro.disconnect()
   189→  }, [draw, fitView])
   190→
   191→  // ---------- загрузка из localStorage ----------
   192→  useEffect(() => {
   193→    try {
   194→      const raw = localStorage.getItem(STORAGE_KEY)
   195→      if (raw) {
   196→        const parsedJson = JSON.parse(raw)
   197→        const parsed = validateSaveFile(parsedJson)
   198→        if (parsed) {
   199→          docRef.current = parsed
   200→          setDoc(parsed)
   201→          if (typeof parsedJson.showGrid === 'boolean') {
   202→            showGridRef.current = parsedJson.showGrid
   203→            setShowGrid(parsedJson.showGrid)
   204→          }
   205→        }
   206→      }
   207→    } catch {
   208→      // повреждённое сохранение — игнорируем
   209→    }
   210→    loadedRef.current = true
   211→  }, [])
   212→
   213→  // ---------- автосохранение ----------
   214→  useEffect(() => {
   215→    if (!loadedRef.current) return
   216→    const t = setTimeout(() => {
   217→      try {
   218→        localStorage.setItem(STORAGE_KEY, JSON.stringify(makeSaveFile(doc, showGrid)))
   219→        setSavedAt(new Date().toLocaleTimeString('ru-RU', { hour: '2-digit', minute: '2-digit' }))
   220→      } catch {
   221→        // нет доступа к localStorage
   222→      }
   223→    }, 500)
   224→    return () => clearTimeout(t)
   225→  }, [doc, showGrid])
   226→
   227→  // ---------- перерисовка при изменении состояния ----------
   228→  useEffect(() => {
   229→    draw()
   230→  }, [doc, showGrid, tool, selectedId, placePreset, draw])
   231→
   232→  // ---------- очистка выделения ----------
   233→  useEffect(() => {
   234→    if (selectedId && !doc.objects.some((o) => o.id === selectedId)) {
   235→      selectedIdRef.current = null
   236→      setSelectedId(null)
   237→    }
   238→  }, [doc, selectedId])
   239→
   240→  // ---------- действия ----------
   241→  const startWallDrawing = useCallback(() => {
   242→    drawingPtsRef.current = []
   243→    cursorPlanRef.current = null
   244→    placePresetRef.current = null
   245→    setPlacePreset(null)
   246→    toolRef.current = 'wall'
   247→    setTool('wall')
   248→    draw()
   249→  }, [draw])
   250→
   251→  const handleToolChange = useCallback(
   252→    (t: Tool) => {
   253→      if (t !== 'wall') drawingPtsRef.current = null
   254→      if (t !== 'select') {
   255→        placePresetRef.current = null
   256→        setPlacePreset(null)
   257→        ghostRef.current = null
   258→      }
   259→      toolRef.current = t
   260→      setTool(t)
   261→      draw()
   262→    },
   263→    [draw],
   264→  )
   265→
   266→  const closeRoom = useCallback(() => {
   267→    const pts = drawingPtsRef.current
   268→    if (!pts) return
   269→    if (pts.length < 3) {
   270→      drawingPtsRef.current = null
   271→      draw()
   272→      return
   273→    }
   274→    pushHistory()
   275→    applyDoc({ ...docRef.current, room: pts })
   276→    drawingPtsRef.current = null
   277→    cursorPlanRef.current = null
   278→    toolRef.current = 'select'
   279→    setTool('select')
   280→    userViewRef.current = false
   281→    fitView()
   282→    draw()
   283→    toast.success('Комната создана — добавьте объекты из каталога')
   284→  }, [applyDoc, pushHistory, fitView, draw])
   285→
   286→  const placeObject = useCallback(
   287→    (plan: Pt, keep: boolean) => {
   288→      const preset = placePresetRef.current
   289→      if (!preset) return
   290→      const step = docRef.current.gridStep
   291→      let x = plan.x
   292→      let y = plan.y
   293→      if (showGridRef.current) {
   294→        const s = snapObjectPos({ x, y, w: preset.w, h: preset.h }, step)
   295→        x = s.x
   296→        y = s.y
   297→      }
   298→      const obj: PlannerObject = {
   299→        id: uid(),
   300→        presetId: preset.id,
   301→        name: preset.name,
   302→        x,
   303→        y,
   304→        w: preset.w,
   305→        h: preset.h,
   306→        angle: 0,
   307→        color: preset.color,
   308→      }
   309→      pushHistory()
   310→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, obj] })
   311→      selectedIdRef.current = obj.id
   312→      setSelectedId(obj.id)
   313→      if (!keep) {
   314→        placePresetRef.current = null
   315→        setPlacePreset(null)
   316→        ghostRef.current = null
   317→      }
   318→    },
   319→    [applyDoc, pushHistory],
   320→  )
   321→
   322→  const deleteObject = useCallback(
   323→    (id: string) => {
   324→      pushHistory()
   325→      applyDoc({ ...docRef.current, objects: docRef.current.objects.filter((o) => o.id !== id) })
   326→      toast('Объект удалён', { icon: '🗑️' })
   327→    },
   328→    [applyDoc, pushHistory],
   329→  )
   330→
   331→  const duplicateObject = useCallback(
   332→    (id: string) => {
   333→      const o = docRef.current.objects.find((x) => x.id === id)
   334→      if (!o) return
   335→      pushHistory()
   336→      const copy: PlannerObject = { ...o, id: uid(), x: o.x + 20, y: o.y + 20 }
   337→      applyDoc({ ...docRef.current, objects: [...docRef.current.objects, copy] })
   338→      selectedIdRef.current = copy.id
   339→      setSelectedId(copy.id)
   340→    },
   341→    [applyDoc, pushHistory],
   342→  )
   343→
   344→  const updateObject = useCallback(
   345→    (id: string, patch: Partial<PlannerObject>) => {
   346→      applyDoc({ ...docRef.current, objects: docRef.current.objects.map((o) => (o.id === id ? { ...o, ...patch } : o)) })
   347→    },
   348→    [applyDoc],
   349→  )
   350→
   351→  const clearRoom = useCallback(() => {
   352→    if (!docRef.current.room) return
   353→    pushHistory()
   354→    applyDoc({ ...docRef.current, room: null })
   355→    toast('Стены удалены — нарисуйте новый контур')
   356→  }, [applyDoc, pushHistory])
   357→
   358→  const handleNew = useCallback(() => {
   359→    pushHistory()
   360→    applyDoc({ ...DEFAULT_DOC, gridStep: docRef.current.gridStep })
   361→    selectedIdRef.current = null
   362→    setSelectedId(null)
   363→    drawingPtsRef.current = null
   364→    ghostRef.current = null
   365→    userViewRef.current = false
   366→    fitView()
   367→    draw()
   368→    toast('Создан новый проект')
   369→  }, [applyDoc, pushHistory, fitView, draw])
   370→
   371→  const handleImportFile = useCallback(
   372→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   373→      const file = e.target.files?.[0]
   374→      e.target.value = ''
   375→      if (!file) return
   376→      try {
   377→        const text = await file.text()
   378→        const parsed = validateSaveFile(JSON.parse(text))
   379→        if (!parsed) throw new Error('bad file')
   380→        pushHistory()
   381→        applyDoc(parsed)
   382→        selectedIdRef.current = null
   383→        setSelectedId(null)
   384→        userViewRef.current = false
   385→        fitView()
   386→        draw()
   387→        toast.success('Проект загружен из файла')
   388→      } catch {
   389→        toast.error('Не удалось прочитать файл проекта')
   390→      }
   391→    },
   392→    [applyDoc, pushHistory, fitView, draw],
   393→  )
   394→
   395→  const handlePickPreset = useCallback(
   396→    (p: Preset) => {
   397→      placePresetRef.current = p
   398→      setPlacePreset(p)
   399→      drawingPtsRef.current = null
   400→      toolRef.current = 'select'
   401→      setTool('select')
   402→      draw()
   403→    },
   404→    [draw],
   405→  )
   406→
   407→  // ---------- слушатели мыши и клавиатуры ----------
   408→  useEffect(() => {
   409→    const canvas = canvasRef.current
   410→    if (!canvas) return
   411→
   412→    const getPos = (e: PointerEvent | WheelEvent) => {
   413→      const r = canvas.getBoundingClientRect()
   414→      return { x: e.clientX - r.left, y: e.clientY - r.top }
   415→    }
   416→
   417→    const updateCoords = (plan: Pt | null) => {
   418→      if (!coordsRef.current) return
   419→      coordsRef.current.textContent = plan
   420→        ? `X: ${Math.round(plan.x)} · Y: ${Math.round(plan.y)} см`
   421→        : '—'
   422→    }
   423→
   424→    const updateCursorStyle = (plan: Pt | null) => {
   425→      if (spaceRef.current || toolRef.current === 'pan' || interRef.current.type === 'pan') {
   426→        canvas.style.cursor = interRef.current.type === 'pan' ? 'grabbing' : 'grab'
   427→        return
   428→      }
   429→      if (toolRef.current === 'wall' || placePresetRef.current) {
   430→        canvas.style.cursor = 'crosshair'
   431→        return
   432→      }
   433→      if (plan) {
   434→        const sel = docRef.current.objects.find((o) => o.id === selectedIdRef.current)
   435→        if (sel) {
   436→          const hp = rotateHandlePos(sel, 26 / viewRef.current.scale)
   437→          if (Math.hypot((hp.x - plan.x) * viewRef.current.scale, (hp.y - plan.y) * viewRef.current.scale) < 10) {
   438→            canvas.style.cursor = 'pointer'
   439→            return
   440→          }
   441→        }
   442→        if (docRef.current.room) {
   443→          for (const p of docRef.current.room) {
   444→            if (Math.hypot((p.x - plan.x) * viewRef.current.scale, (p.y - plan.y) * viewRef.current.scale) < 9) {
   445→              canvas.style.cursor = 'pointer'
   446→              return
   447→            }
   448→          }
   449→        }
   450→        for (let i = docRef.current.objects.length - 1; i >= 0; i--) {
   451→          if (pointInObject(docRef.current.objects[i], plan)) {
   452→            canvas.style.cursor = 'move'
   453→            return
   454→          }
   455→        }
   456→      }
   457→      canvas.style.cursor = 'default'
   458→    }
   459→
   460→    const zoomAt = (px: number, py: number, factor: number) => {
   461→      const v = viewRef.current
   462→      const ns = Math.min(MAX_SCALE, Math.max(MIN_SCALE, v.scale * factor))
   463→      const plan = screenToPlan(px, py, v)
   464→      viewRef.current = { scale: ns, ox: px - plan.x * ns, oy: py - plan.y * ns }
   465→      userViewRef.current = true
   466→      draw()
   467→    }
   468→
   469→    const onPointerDown = (e: PointerEvent) => {
   470→      const pos = getPos(e)
   471→      const plan = screenToPlan(pos.x, pos.y, viewRef.current)
   472→
   473→      // панорамирование: средняя кнопка, пробел или инструмент «Рука»
   474→      if (e.button === 1 || spaceRef.current || (toolRef.current === 'pan' && e.button === 0)) {
   475→        e.preventDefault()
   476→        interRef.current = {
   477→          type: 'pan',
   478→          panStart: { ox: viewRef.current.ox, oy: viewRef.current.oy, px: pos.x, py: pos.y },
   479→        }
   480→        canvas.style.cursor = 'grabbing'
   481→        return
   482→      }
   483→
   484→      if (e.button === 2) {
   485→        // ПКМ в режиме стен — убрать последнюю точку
   486→        if (toolRef.current === 'wall' && drawingPtsRef.current && drawingPtsRef.current.length > 0) {
   487→          e.preventDefault()
   488→          const pts = [...drawingPtsRef.current]
   489→          pts.pop()
   490→          drawingPtsRef.current = pts
   491→          draw()
   492→        }
   493→        return
   494→      }
   495→      if (e.button !== 0) return
   496→
   497→      // рисование стен
   498→      if (toolRef.current === 'wall') {
   499→        const pts = drawingPtsRef.current ?? []
   500→        if (pts.length >= 3) {
   501→          const f = pts[0]
   502→          if (Math.hypot(plan.x - f.x, plan.y - f.y) * viewRef.current.scale < 12) {
   503→            closeRoom()
   504→            return
   505→          }
   506→        }
   507→        const step = docRef.current.gridStep
   508→        drawingPtsRef.current = [...pts, { x: snapValue(plan.x, step), y: snapValue(plan.y, step) }]
   509→        draw()
   510→        return
   511→      }
   512→
   513→      // размещение объекта
   514→      if (placePresetRef.current) {
   515→        placeObject(plan, e.shiftKey)
   516→        return
   517→      }
   518→
   519→      // инструмент выбора
   520→      const sel = docRef.current.objects.find((o) => o.id === selectedIdRef.current)
   521→      if (sel) {
   522→        const hp = rotateHandlePos(sel, 26 / viewRef.current.scale)
   523→        if (Math.hypot((hp.x - plan.x) * viewRef.current.scale, (hp.y - plan.y) * viewRef.current.scale) < 10) {
   524→          pushHistory()
   525→          interRef.current = { type: 'rotate' }
   526→          canvas.style.cursor = 'grabbing'
   527→          return
   528→        }
   529→      }
   530→      if (docRef.current.room) {
   531→        const room = docRef.current.room
   532→        for (let i = 0; i < room.length; i++) {
   533→          if (Math.hypot((room[i].x - plan.x) * viewRef.current.scale, (room[i].y - plan.y) * viewRef.current.scale) < 9) {
   534→            pushHistory()
   535→            interRef.current = { type: 'vertex', vertexIdx: i }
   536→            dragVertexRef.current = i
   537→            return
   538→          }
   539→        }
   540→      }
   541→      for (let i = docRef.current.objects.length - 1; i >= 0; i--) {
   542→        const o = docRef.current.objects[i]
   543→        if (pointInObject(o, plan)) {
   544→          selectedIdRef.current = o.id
   545→          setSelectedId(o.id)
   546→          pushHistory()
   547→          interRef.current = { type: 'drag', grabDX: plan.x - o.x, grabDY: plan.y - o.y, moved: false }
   548→          canvas.style.cursor = 'grabbing'
   549→          return
   550→        }
   551→      }
   552→      // пустое место — снять выделение
   553→      selectedIdRef.current = null
   554→      setSelectedId(null)
   555→    }
   556→
   557→    const onPointerMove = (e: PointerEvent) => {
   558→      const pos = getPos(e)
   559→      const plan = screenToPlan(pos.x, pos.y, viewRef.current)
   560→      cursorPlanRef.current = plan
   561→      updateCoords(plan)
   562→      const it = interRef.current
   563→
   564→      if (it.type === 'pan' && it.panStart) {
   565→        viewRef.current = {
   566→          ...viewRef.current,
   567→          ox: it.panStart.ox + (pos.x - it.panStart.px),
   568→          oy: it.panStart.oy + (pos.y - it.panStart.py),
   569→        }
   570→        userViewRef.current = true
   571→        draw()
   572→        return
   573→      }
   574→
   575→      if (it.type === 'drag') {
   576→        const o = docRef.current.objects.find((x) => x.id === selectedIdRef.current)
   577→        if (!o || it.grabDX === undefined) return
   578→        let nx = plan.x - it.grabDX
   579→        let ny = plan.y - it.grabDY
   580→        if (showGridRef.current && !e.altKey) {
   581→          const s = snapObjectPos({ x: nx, y: ny, w: o.w, h: o.h }, docRef.current.gridStep)
   582→          nx = s.x
   583→          ny = s.y
   584→        }
   585→        it.moved = true
   586→        applyDoc({ ...docRef.current, objects: docRef.current.objects.map((x) => (x.id === o.id ? { ...x, x: nx, y: ny } : x)) })
   587→        return
   588→      }
   589→
   590→      if (it.type === 'rotate') {
   591→        const o = docRef.current.objects.find((x) => x.id === selectedIdRef.current)
   592→        if (!o) return
   593→        let angle = (Math.atan2(plan.y - o.y, plan.x - o.x) * 180) / Math.PI + 90
   594→        if (!e.altKey) angle = Math.round(angle / 15) * 15
   595→        angle = ((((angle + 180) % 360) + 360) % 360) - 180
   596→        applyDoc({ ...docRef.current, objects: docRef.current.objects.map((x) => (x.id === o.id ? { ...x, angle } : x)) })
   597→        return
   598→      }
   599→
   600→      if (it.type === 'vertex' && it.vertexIdx !== undefined && docRef.current.room) {
   601→        const idx = it.vertexIdx
   602→        const step = docRef.current.gridStep
   603→        const np = e.altKey ? { x: plan.x, y: plan.y } : { x: snapValue(plan.x, step), y: snapValue(plan.y, step) }
   604→        const room = docRef.current.room.map((p, i) => (i === idx ? np : p))
   605→        applyDoc({ ...docRef.current, room })
   606→        return
   607→      }
   608→
   609→      // без зажатой кнопки: призрак и курсор
   610→      if (placePresetRef.current) {
   611→        const p = placePresetRef.current
   612→        let gx = plan.x
   613→        let gy = plan.y
   614→        if (showGridRef.current) {
   615→          const s = snapObjectPos({ x: gx, y: gy, w: p.w, h: p.h }, docRef.current.gridStep)
   616→          gx = s.x
   617→          gy = s.y
   618→        }
   619→        ghostRef.current = { id: 'ghost', presetId: p.id, name: p.name, x: gx, y: gy, w: p.w, h: p.h, angle: 0, color: p.color }
   620→        draw()
   621→      } else if (toolRef.current === 'wall' && drawingPtsRef.current) {
   622→        draw()
   623→      }
   624→      updateCursorStyle(plan)
   625→    }
   626→
   627→    const onPointerUp = () => {
   628→      const it = interRef.current
   629→      if (it.type === 'drag' && !it.moved) {
   630→        // клик без перемещения — убрать пустой шаг истории
   631→        historyRef.current.past.pop()
   632→        setHist({ p: historyRef.current.past.length, f: historyRef.current.future.length })
   633→      }
   634→      if (it.type === 'vertex') dragVertexRef.current = null
   635→      interRef.current = { type: 'none' }
   636→      updateCursorStyle(cursorPlanRef.current)
   637→    }
   638→
   639→    const onWheel = (e: WheelEvent) => {
   640→      e.preventDefault()
   641→      const pos = getPos(e)
   642→      const factor = Math.exp(-e.deltaY * 0.0012)
   643→      zoomAt(pos.x, pos.y, factor)
   644→    }
   645→
   646→    const onDoubleClick = () => {
   647→      if (toolRef.current === 'wall' && drawingPtsRef.current && drawingPtsRef.current.length >= 3) {
   648→        closeRoom()
   649→      }
   650→    }
   651→
   652→    const onContextMenu = (e: MouseEvent) => e.preventDefault()
   653→
   654→    const onKeyDown = (e: KeyboardEvent) => {
   655→      const target = e.target as HTMLElement | null
   656→      if (target && (['INPUT', 'TEXTAREA', 'SELECT'].includes(target.tagName) || target.isContentEditable)) return
   657→
   658→      if (e.code === 'Space') {
   659→        if (!spaceRef.current) spaceRef.current = true
   660→        e.preventDefault()
   661→        updateCursorStyle(cursorPlanRef.current)
   662→        return
   663→      }
   664→
   665→      const mod = e.ctrlKey || e.metaKey
   666→
   667→      if (mod && e.code === 'KeyZ') {
   668→        e.preventDefault()
   669→        if (e.shiftKey) redo()
   670→        else undo()
   671→        return
   672→      }
   673→      if (mod && e.code === 'KeyY') {
   674→        e.preventDefault()
   675→        redo()
   676→        return
   677→      }
   678→      if (mod && e.code === 'KeyD') {
   679→        e.preventDefault()
   680→        if (selectedIdRef.current) duplicateObject(selectedIdRef.current)
   681→        return
   682→      }
   683→
   684→      if (e.key === 'Escape') {
   685→        if (drawingPtsRef.current) {
   686→          drawingPtsRef.current = null
   687→          draw()
   688→        } else if (placePresetRef.current) {
   689→          placePresetRef.current = null
   690→          setPlacePreset(null)
   691→          ghostRef.current = null
   692→          draw()
   693→        } else if (selectedIdRef.current) {
   694→          selectedIdRef.current = null
   695→          setSelectedId(null)
   696→        }
   697→        return
   698→      }
   699→
   700→      if (e.key === 'Enter' && drawingPtsRef.current) {
   701→        closeRoom()
   702→        return
   703→      }
   704→
   705→      if ((e.key === 'Delete' || e.key === 'Backspace') && selectedIdRef.current) {
   706→        e.preventDefault()
   707→        deleteObject(selectedIdRef.current)
   708→        return
   709→      }
   710→
   711→      if (e.code === 'KeyR' && selectedIdRef.current) {
   712→        const o = docRef.current.objects.find((x) => x.id === selectedIdRef.current)
   713→        if (o) {
   714→          pushHistory()
   715→          const d = e.shiftKey ? -90 : 90
   716→          updateObject(o.id, { angle: ((((o.angle + d + 180) % 360) + 360) % 360) - 180 })
   717→        }
   718→        return
   719→      }
   720→
   721→      if (e.code === 'KeyV' || e.code === 'Digit1') handleToolChange('select')
   722→      if (e.code === 'KeyW' || e.code === 'Digit2') handleToolChange('wall')
   723→      if (e.code === 'KeyH' || e.code === 'Digit3') handleToolChange('pan')
   724→
   725→      // стрелки — точное перемещение
   726→      if (selectedIdRef.current && e.key.startsWith('Arrow')) {
   727→        e.preventDefault()
   728→        const o = docRef.current.objects.find((x) => x.id === selectedIdRef.current)
   729→        if (!o) return
   730→        const base = showGridRef.current ? docRef.current.gridStep : 10
   731→        const step = e.shiftKey ? 1 : base
   732→        let dx = 0
   733→        let dy = 0
   734→        if (e.key === 'ArrowLeft') dx = -step
   735→        if (e.key === 'ArrowRight') dx = step
   736→        if (e.key === 'ArrowUp') dy = -step
   737→        if (e.key === 'ArrowDown') dy = step
   738→        const now = Date.now()
   739→        if (now - lastNudgeRef.current > 600) pushHistory()
   740→        lastNudgeRef.current = now
   741→        applyDoc({ ...docRef.current, objects: docRef.current.objects.map((x) => (x.id === o.id ? { ...x, x: o.x + dx, y: o.y + dy } : x)) })
   742→      }
   743→    }
   744→
   745→    const onKeyUp = (e: KeyboardEvent) => {
   746→      if (e.code === 'Space') {
   747→        spaceRef.current = false
   748→        updateCursorStyle(cursorPlanRef.current)
   749→      }
   750→    }
   751→
   752→    canvas.addEventListener('pointerdown', onPointerDown)
   753→    window.addEventListener('pointermove', onPointerMove)
   754→    window.addEventListener('pointerup', onPointerUp)
   755→    canvas.addEventListener('wheel', onWheel, { passive: false })
   756→    canvas.addEventListener('dblclick', onDoubleClick)
   757→    canvas.addEventListener('contextmenu', onContextMenu)
   758→    window.addEventListener('keydown', onKeyDown)
   759→    window.addEventListener('keyup', onKeyUp)
   760→
   761→    return () => {
   762→      canvas.removeEventListener('pointerdown', onPointerDown)
   763→      window.removeEventListener('pointermove', onPointerMove)
   764→      window.removeEventListener('pointerup', onPointerUp)
   765→      canvas.removeEventListener('wheel', onWheel)
   766→      canvas.removeEventListener('dblclick', onDoubleClick)
   767→      canvas.removeEventListener('contextmenu', onContextMenu)
   768→      window.removeEventListener('keydown', onKeyDown)
   769→      window.removeEventListener('keyup', onKeyUp)
   770→    }
   771→  }, [
   772→    applyDoc,
   773→    closeRoom,
   774→    deleteObject,
   775→    duplicateObject,
   776→    draw,
   777→    fitView,
   778→    handleToolChange,
   779→    placeObject,
   780→    pushHistory,
   781→    redo,
   782→    undo,
   783→    updateObject,
   784→  ])
   785→
   786→  const selected = doc.objects.find((o) => o.id === selectedId) ?? null
   787→  const empty = !doc.room && doc.objects.length === 0
   788→
   789→  const hint = tool === 'wall'
   790→    ? 'Кликайте по углам комнаты · Enter или клик по первой точке — замкнуть · ПКМ — убрать точку · Esc — отмена'
   791→    : placePreset
   792→      ? `Размещение: ${placePreset.name} — кликните на плане · Shift+клик — несколько · Esc — отмена`
   793→      : null
   794→
   795→  return (
   796→    <div className="flex h-[100dvh] min-h-0 flex-col bg-[#F6F1E9] text-[#3D3428]">
   797→      <TopBar
   798→        tool={tool}
   799→        setTool={handleToolChange}
   800→        canUndo={hist.p > 0}
   801→        canRedo={hist.f > 0}
   802→        onUndo={undo}
   803→        onRedo={redo}
   804→        onFit={() => {
   805→          userViewRef.current = false
   806→          fitView()
   807→          draw()
   808→        }}
   809→        onZoomIn={() => {
   810→          zoomCenter(1.25)
   811→        }}
   812→        onZoomOut={() => {
   813→          zoomCenter(0.8)
   814→        }}
   815→        savedAt={savedAt}
   816→        onNew={handleNew}
   817→        onImportClick={() => fileInputRef.current?.click()}
   818→        onExportPNG={() => {
   819→          exportPNG(docRef.current)
   820→          toast.success('PNG-файл скачивается…')
   821→        }}
   822→        onExportJSON={() => {
   823→          exportJSON(docRef.current, showGridRef.current)
   824→          toast.success('JSON-файл скачивается…')
   825→        }}
   826→      />
   827→
   828→      <div className="flex min-h-0 flex-1 flex-col lg:flex-row">
   829→        {/* Каталог */}
   830→        <aside className="order-2 max-h-52 w-full shrink-0 overflow-hidden border-t border-[#E7DECF] bg-[#FBF7EF] lg:order-1 lg:max-h-none lg:w-52 lg:border-t-0 lg:border-r xl:w-60">
   831→          <Catalog placePreset={placePreset} onPlace={handlePickPreset} />
   832→        </aside>
   833→
   834→        {/* Холст */}
   835→        <main className="relative order-1 min-h-[44vh] flex-1 lg:order-2 lg:min-h-0" aria-label="План помещения">
   836→          <div ref={wrapRef} className="absolute inset-0">
   837→            <canvas ref={canvasRef} className="absolute inset-0 h-full w-full touch-none select-none" />
   838→          </div>
   839→
   840→          {hint && (
   841→            <div className="pointer-events-none absolute top-3 left-1/2 z-10 max-w-[92%] -translate-x-1/2 rounded-full bg-[#3D3428]/85 px-4 py-1.5 text-center text-[11px] font-medium text-[#FBF7EF] shadow-lg">
   842→              {hint}
   843→            </div>
   844→          )}
   845→
   846→          {empty && !hint && (
   847→            <div className="pointer-events-none absolute inset-0 flex items-center justify-center p-4">
   848→              <div className="pointer-events-auto max-w-sm rounded-2xl border border-[#EAE0CE] bg-[#FBF7EF]/95 p-6 text-center shadow-xl">
   849→                <div className="mx-auto mb-3 flex h-12 w-12 items-center justify-center rounded-2xl bg-[#F4E8D4]">
   850→                  <Ruler className="h-6 w-6 text-[#E8730C]" />
   851→                </div>
   852→                <h2 className="text-lg font-bold text-[#3D3428]">Нарисуйте планировку</h2>
   853→                <ol className="mt-2 space-y-1 text-left text-xs leading-relaxed text-[#6B5D4F]">
   854→                  <li>
   855→                    <b>1.</b> Инструментом «Стены» кликните углы комнаты в метрах от угла — контур замкнётся автоматически.
   856→                  </li>
   857→                  <li>
   858→                    <b>2.</b> Выберите мебель в каталоге и кликните на плане, чтобы поставить.
   859→                  </li>
   860→                  <li>
   861→                    <b>3.</b> Перетаскивайте объекты мышью, вращайте за оранжевую ручку, задавайте точные размеры справа.
   862→                  </li>
   863→                </ol>
   864→                <Button className="mt-4 h-9 bg-[#E8730C] text-sm font-semibold text-white shadow-sm hover:bg-[#D4660A]" onClick={startWallDrawing}>
   865→                  <PencilLine className="mr-1.5 h-4 w-4" /> Нарисовать стены
   866→                </Button>
   867→              </div>
   868→            </div>
   869→          )}
   870→        </main>
   871→
   872→        {/* Свойства */}
   873→        <aside className="order-3 max-h-60 w-full shrink-0 overflow-hidden border-t border-[#E7DECF] bg-[#FBF7EF] lg:max-h-none lg:w-72 lg:border-t-0 lg:border-l">
   874→          <PropertiesPanel
   875→            doc={doc}
   876→            showGrid={showGrid}
   877→            selected={selected}
   878→            onUpdateObject={updateObject}
   879→            onCommit={() => pushHistory()}
   880→            onDeleteObject={deleteObject}
   881→            onDuplicateObject={duplicateObject}
   882→            onClearRoom={clearRoom}
   883→            onGridStepChange={(s) => applyDoc({ ...docRef.current, gridStep: s })}
   884→            onToggleGrid={(v) => setShowGrid(v)}
   885→          />
   886→        </aside>
   887→      </div>
   888→
   889→      {/* Статусная строка */}
   890→      <footer className="z-10 flex h-8 shrink-0 items-center justify-between border-t border-[#E7DECF] bg-[#FBF7EF] px-3 text-[11px] font-medium text-[#8B7D6B] sm:px-4">
   891→        <span className="truncate">
   892→          {tool === 'wall' ? 'Режим: рисование стен' : tool === 'pan' ? 'Режим: перетаскивание холста' : 'Режим: выбор и редактирование'} · колесо — масштаб, пробел — панорама
   893→        </span>
   894→        <div className="flex shrink-0 items-center gap-3 tabular-nums sm:gap-4">
   895→          <span ref={coordsRef} className="hidden sm:inline">
   896→            —
   897→          </span>
   898→          <span>
   899→            Масштаб: <span ref={zoomRef}>—</span>
   900→          </span>
   901→          <span className="hidden md:inline">Объектов: {doc.objects.length}</span>
   902→        </div>
   903→      </footer>
   904→
   905→      <input ref={fileInputRef} type="file" accept=".json,application/json" className="hidden" onChange={handleImportFile} />
   906→    </div>
   907→  )
